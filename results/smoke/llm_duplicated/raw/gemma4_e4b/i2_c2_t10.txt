```sql
DECLARE @SubjectID INT = 42;
DECLARE @FullName NVARCHAR(100) = 'Subject_0000042';
DECLARE @Email NVARCHAR(100) = 'subject42@example.invalid';
DECLARE @Phone NVARCHAR(50) = '+1-555-0000042';

-- Check 1: The current version of the customer table (dbo.customer)
SELECT 
    'dbo.customer' AS object_name, 
    COUNT(1) AS rows_found
UNION ALL
-- Check 2: The history table where deleted records are moved (dbo.customer_history)
SELECT 
    'dbo.customer_history' AS object_name, 
    COUNT(1)
FROM dbo.customer_history
WHERE customer_id = @SubjectID
  AND full_name = @FullName
  AND email = @Email
  AND phone = @Phone;
```
