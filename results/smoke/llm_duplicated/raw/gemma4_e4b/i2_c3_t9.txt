```sql
DECLARE @TargetFullName NVARCHAR(100) = 'Subject_0000042';
DECLARE @TargetEmail NVARCHAR(100) = 'subject42@example.invalid';
DECLARE @TargetPhone NVARCHAR(50) = '+1-555-0000042';

-- Check 1: Primary System Versioned Table (dbo.customer)
SELECT
    'dbo.customer' AS object_name,
    COUNT(1) AS rows_found
FROM dbo.customer
WHERE full_name = @TargetFullName OR email = @TargetEmail OR phone = @TargetPhone

UNION ALL

-- Check 2: Temporal History Table (dbo.customer_history)
SELECT
    'dbo.customer_history' AS object_name,
    COUNT(1) AS rows_found
FROM dbo.customer_history
WHERE full_name = @TargetFullName OR email = @TargetEmail OR phone = @TargetPhone;
```
