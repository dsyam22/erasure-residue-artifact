```sql
WITH ObjectChecks AS (
    -- 1. Check the current/active table version
    SELECT
        'dbo.customer' AS object_name,
        COUNT(CASE WHEN customer_id = 42 OR full_name LIKE '%Subject_0000042%' OR email = 'subject42@example.invalid' OR phone = '+1-555-0000042' THEN 1 END) AS rows_found
    FROM dbo.customer

    UNION ALL

    -- 2. Check the temporal history table (where deleted records reside)
    SELECT
        'dbo.customer_history' AS object_name,
        COUNT(CASE WHEN customer_id = 42 OR full_name LIKE '%Subject_0000042%' OR email = 'subject42@example.invalid' OR phone = '+1-555-0000042' THEN 1 END) AS rows_found
    FROM dbo.customer_history

    -- Note: Since CDC is disabled and no other tables were specified, these are the only locations checked.
)
SELECT
    object_name,
    rows_found
FROM ObjectChecks;
```
