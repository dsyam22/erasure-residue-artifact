    FROM dbo.customer
) AS T1
UNION ALL
SELECT 'dbo.customer_history' AS object_name, ISNULL(T2.rows_found, 0) FROM (
    SELECT COUNT(CASE WHEN customer_id = 42 AND full_name = N'Subject_0000042' AND email = N'subject42@example.invalid' AND phone = N'+1-555-0000042' THEN 1 END) AS rows_found
    FROM dbo.customer_history
) AS T2;
