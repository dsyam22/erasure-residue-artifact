```sql
-- SCRIPT
SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

DECLARE @results TABLE (
    object_name sysname,
    rows_found bigint
);

-- Check dbo.customer
INSERT INTO @results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check other tables in the database that might contain the personal data
DECLARE @sql nvarchar(max) = N'';
DECLARE @schema sysname, @table sysname, @column sysname;

DECLARE cur CURSOR LOCAL FAST_FORWARD FOR
SELECT c.TABLE_SCHEMA, c.TABLE_NAME, c.COLUMN_NAME
FROM INFORMATION_SCHEMA.COLUMNS c
INNER JOIN INFORMATION_SCHEMA.TABLES t
    ON c.TABLE_SCHEMA = t.TABLE_SCHEMA AND c.TABLE_NAME = t.TABLE_NAME
WHERE t.TABLE_TYPE = 'BASE TABLE'
  AND c.DATA_TYPE IN ('nvarchar', 'varchar', 'nchar', 'char', 'ntext', 'text')
  AND NOT (c.TABLE_SCHEMA = 'dbo' AND c.TABLE_NAME = 'customer');

OPEN cur;
FETCH NEXT FROM cur INTO @schema, @table, @column;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @sql = @sql + N'
    INSERT INTO @results (object_name, rows_found)
    SELECT N''' + REPLACE(@schema + '.' + @table, '''', '''''') + N''', COUNT(*)
    FROM [' + @schema + N'].[' + @table + N']
    WHERE [' + @column + N'] IN (@full_name, @email, @phone)
       OR CAST([' + @column + N'] AS nvarchar(4000)) LIKE N''%Subject_0000042%''
       OR CAST([' + @column + N'] AS nvarchar(4000)) LIKE N''%subject42@example.invalid%''
       OR CAST([' + @column + N'] AS nvarchar(4000)) LIKE N''%+1-555-0000042%'';' + NCHAR(13) + NCHAR(10);

    FETCH NEXT FROM cur INTO @schema, @table, @column;
END

CLOSE cur;
DEALLOCATE cur;

IF @sql <> N''
BEGIN
    EXEC sp_executesql @sql,
        N'@full_name nvarchar(200), @email nvarchar(320), @phone nvarchar(50), @results TABLE (object_name sysname, rows_found bigint) OUTPUT',
        @full_name, @email, @phone, @results;
END

SELECT object_name, rows_found
FROM @results
ORDER BY object_name;
```
