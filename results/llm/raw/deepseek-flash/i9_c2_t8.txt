/* Read-only verification: does the erased subject's personal data still exist?
   Checks dbo.customer (current) and dbo.customer_history (temporal history).
   Returns exactly one result set: object_name, rows_found. */

SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @full_name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(200) = N'subject42@example.invalid';
DECLARE @phone nvarchar(200) = N'+1-555-0000042';

;WITH checks AS
(
    SELECT
        N'dbo.customer' AS object_name,
        COUNT(*) AS rows_found
    FROM dbo.customer
    WHERE customer_id = @cid
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone

    UNION ALL

    SELECT
        N'dbo.customer_history' AS object_name,
        COUNT(*) AS rows_found
    FROM dbo.customer_history
    WHERE customer_id = @cid
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone
)
SELECT object_name, rows_found
FROM checks
ORDER BY object_name;
