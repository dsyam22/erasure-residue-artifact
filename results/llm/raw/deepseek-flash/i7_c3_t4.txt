/* Read-only verification script: checks whether the erased subject's original
   personal data still exists anywhere in the database.
   Returns exactly one result set: object_name, rows_found. */

SET NOCOUNT ON;

DECLARE @cid    int           = 42;
DECLARE @fname  nvarchar(200) = N'Subject_0000042';
DECLARE @email  nvarchar(200) = N'subject42@example.invalid';
DECLARE @phone  nvarchar(200) = N'+1-555-0000042';

/* Collect results from every object we inspect. */
CREATE TABLE #results (object_name sysname NOT NULL, rows_found bigint NOT NULL);

/* ------------------------------------------------------------------
   1) Base table dbo.customer
   ------------------------------------------------------------------ */
INSERT INTO #results (object_name, rows_found)
SELECT 'dbo.customer',
       COUNT(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name   = @fname
   OR email       = @email
   OR phone       = @phone;

/* ------------------------------------------------------------------
   2) Downstream copy reporting.customer_replica
   ------------------------------------------------------------------ */
INSERT INTO #results (object_name, rows_found)
SELECT 'reporting.customer_replica',
       COUNT(*)
FROM reporting.customer_replica
WHERE customer_id = @cid
   OR full_name   = @fname
   OR email       = @email
   OR phone       = @phone;

/* ------------------------------------------------------------------
   3) Temporal history tables (if any table in the DB is system-versioned,
      scan its history table).  None are configured here, but we check
      dynamically so the result set always reports what was inspected.
   ------------------------------------------------------------------ */
DECLARE @tbl sysname, @hist sysname, @sql nvarchar(max);

DECLARE hist_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name + '.' + t.name, s.name + '.' + h.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    JOIN sys.tables AS h ON h.object_id = t.history_table_id
    WHERE t.temporal_type = 1;               /* SYSTEM_VERSIONED_TEMPORAL_TABLE */

OPEN hist_cur;
FETCH NEXT FROM hist_cur INTO @tbl, @hist;
WHILE @@FETCH_STATUS = 0
BEGIN
    SET @sql = N'
        INSERT INTO #results (object_name, rows_found)
        SELECT @obj, COUNT(*)
        FROM ' + QUOTENAME(PARSENAME(@hist,2)) + N'.' + QUOTENAME(PARSENAME(@hist,1)) + N'
        WHERE customer_id = @cid
           OR full_name   = @fname
           OR email       = @email
           OR phone       = @phone;';

    EXEC sp_executesql @sql,
        N'@obj sysname, @cid int, @fname nvarchar(200), @email nvarchar(200), @phone nvarchar(200)',
        @obj = @hist, @cid = @cid, @fname = @fname, @email = @email, @phone = @phone;

    FETCH NEXT FROM hist_cur INTO @tbl, @hist;
END
CLOSE hist_cur;
DEALLOCATE hist_cur;

/* ------------------------------------------------------------------
   4) CDC change tables (database cdc_enabled = 0, so normally none).
      If CDC were on, the capture instances would live in schema cdc.
   ------------------------------------------------------------------ */
DECLARE @capture sysname;

DECLARE cdc_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT QUOTENAME(OBJECT_SCHEMA_NAME(source_object_id)) + '.' + QUOTENAME(object_name)
    FROM cdc.change_tables;

OPEN cdc_cur;
FETCH NEXT FROM cdc_cur INTO @capture;
WHILE @@FETCH_STATUS = 0
BEGIN
    /* CDC capture tables expose the source columns; scan the common ones. */
    SET @sql = N'
        IF EXISTS (SELECT 1 FROM sys.columns
                   WHERE object_id = OBJECT_ID(@obj)
                     AND name IN (''customer_id'',''full_name'',''email'',''phone''))
        BEGIN
            INSERT INTO #results (object_name, rows_found)
            SELECT @obj, COUNT(*)
            FROM ' + @capture + N'
            WHERE customer_id = @cid
               OR full_name   = @fname
               OR email       = @email
               OR phone       = @phone;
        END
        ELSE
        BEGIN
            INSERT INTO #results (object_name, rows_found)
            VALUES (@obj, 0);
        END';

    EXEC sp_executesql @sql,
        N'@obj sysname, @cid int, @fname nvarchar(200), @email nvarchar(200), @phone nvarchar(200)',
        @obj = @capture, @cid = @cid, @fname = @fname, @email = @email, @phone = @phone;

    FETCH NEXT FROM cdc_cur INTO @capture;
END
CLOSE cdc_cur;
DEALLOCATE cdc_cur;

/* ------------------------------------------------------------------
   Final result set: one row per inspected object.
   ------------------------------------------------------------------ */
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
