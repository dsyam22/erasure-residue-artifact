```sql
-- SCRIPT
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @full nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

CREATE TABLE #checks (object_name sysname NOT NULL, rows_found bigint NOT NULL);

INSERT INTO #checks
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @full
   OR email = @email
   OR phone = @phone;

SELECT object_name, rows_found
FROM #checks
ORDER BY object_name;
```
