```sql
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(400) = N'subject42@example.invalid';
DECLARE @phone nvarchar(60) = N'+1-555-0000042';

SELECT object_name, rows_found
FROM (
    SELECT 'dbo.customer' AS object_name,
           (SELECT COUNT(*) FROM dbo.customer
            WHERE customer_id = @cid
               OR full_name = @name
               OR email = @email
               OR phone = @phone) AS rows_found
    UNION ALL
    SELECT 'dbo.member',
           (SELECT COUNT(*) FROM dbo.member
            WHERE member_id = @cid
               OR full_name = @name
               OR email = @email
               OR phone = @phone)
    UNION ALL
    SELECT 'dbo.member_history',
           (SELECT COUNT(*) FROM dbo.member_history
            WHERE member_id = @cid
               OR full_name = @name
               OR email = @email
               OR phone = @phone)
    UNION ALL
    SELECT 'cdc.dbo_customer_CT',
           (SELECT COUNT(*) FROM cdc.dbo_customer_CT
            WHERE customer_id = @cid
               OR full_name = @name
               OR email = @email
               OR phone = @phone)
) AS checks
ORDER BY object_name;
```
