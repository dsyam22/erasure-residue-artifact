/* Read-only verification script: checks whether the subject's original personal data
   still exists anywhere in database llm_i7.
   Returns exactly one result set: object_name, rows_found. */

SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name  nvarchar(200) = N'Subject_0000042';
DECLARE @email      nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone      nvarchar(50)  = N'+1-555-0000042';

DECLARE @results TABLE (object_name sysname NOT NULL, rows_found bigint NOT NULL);

/* 1) Live base table */
INSERT INTO @results (object_name, rows_found)
SELECT N'dbo.customer',
       COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* 2) Downstream replica table */
INSERT INTO @results (object_name, rows_found)
SELECT N'reporting.customer_replica',
       COUNT(*)
FROM reporting.customer_replica
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* 3) CDC change tables (only if CDC is enabled on the database) */
IF EXISTS (SELECT 1 FROM sys.databases WHERE name = DB_NAME() AND is_cdc_enabled = 1)
BEGIN
    DECLARE @schema sysname, @table sysname, @capture_instance sysname;
    DECLARE @sql nvarchar(max);

    DECLARE cdc_cur CURSOR LOCAL FAST_FORWARD FOR
        SELECT OBJECT_SCHEMA_NAME(ct.object_id), OBJECT_NAME(ct.object_id), ct.capture_instance
        FROM cdc.change_tables AS ct;

    OPEN cdc_cur;
    FETCH NEXT FROM cdc_cur INTO @schema, @table, @capture_instance;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        SET @sql = N'
            INSERT INTO @results (object_name, rows_found)
            SELECT N''cdc.'' + @ci,
                   COUNT(*)
            FROM cdc.[' + @capture_instance + N'] AS c
            WHERE c.customer_id = @cid
               OR c.full_name = @fn
               OR c.email = @em
               OR c.phone = @ph;';

        EXEC sp_executesql @sql,
            N'@ci sysname, @cid int, @fn nvarchar(200), @em nvarchar(320), @ph nvarchar(50)',
            @ci = @capture_instance, @cid = @customer_id, @fn = @full_name,
            @em = @email, @ph = @phone;

        FETCH NEXT FROM cdc_cur INTO @schema, @table, @capture_instance;
    END

    CLOSE cdc_cur;
    DEALLOCATE cdc_cur;
END
ELSE
BEGIN
    /* CDC disabled: report the CDC check as 0 rows found */
    INSERT INTO @results (object_name, rows_found)
    SELECT N'cdc.change_tables', 0;
END

/* 4) Temporal history tables (only if any exist for the checked tables) */
IF EXISTS (SELECT 1 FROM sys.tables WHERE temporal_type = 1)
BEGIN
    DECLARE @hist_schema sysname, @hist_table sysname;
    DECLARE @hsql nvarchar(max);

    DECLARE hist_cur CURSOR LOCAL FAST_FORWARD FOR
        SELECT OBJECT_SCHEMA_NAME(history_table_id), OBJECT_NAME(history_table_id)
        FROM sys.tables
        WHERE temporal_type = 1;

    OPEN hist_cur;
    FETCH NEXT FROM hist_cur INTO @hist_schema, @hist_table;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        SET @hsql = N'
            INSERT INTO @results (object_name, rows_found)
            SELECT N''' + @hist_schema + N'.' + @hist_table + N''',
                   COUNT(*)
            FROM [' + @hist_schema + N'].[' + @hist_table + N']
            WHERE customer_id = @cid
               OR full_name = @fn
               OR email = @em
               OR phone = @ph;';

        EXEC sp_executesql @hsql,
            N'@cid int, @fn nvarchar(200), @em nvarchar(320), @ph nvarchar(50)',
            @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone;

        FETCH NEXT FROM hist_cur INTO @hist_schema, @hist_table;
    END

    CLOSE hist_cur;
    DEALLOCATE hist_cur;
END
ELSE
BEGIN
    /* No temporal history tables: report the check as 0 rows found */
    INSERT INTO @results (object_name, rows_found)
    SELECT N'temporal.history_tables', 0;
END

/* Final single result set */
SELECT object_name, rows_found
FROM @results
ORDER BY object_name;
