```sql
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @full_name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

DECLARE @results TABLE (object_name sysname, rows_found bigint);

/* dbo.customer */
INSERT INTO @results (object_name, rows_found)
SELECT N'dbo.customer', COUNT_BIG(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* reporting.customer_replica */
INSERT INTO @results (object_name, rows_found)
SELECT N'reporting.customer_replica', COUNT_BIG(*)
FROM reporting.customer_replica
WHERE customer_id = @cid
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* CDC change tables (if any exist for dbo.customer) */
IF OBJECT_ID(N'cdc.dbo_customer_CT', N'U') IS NOT NULL
BEGIN
    INSERT INTO @results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', COUNT_BIG(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @cid
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END

/* Temporal history table (if dbo.customer were temporal) */
IF OBJECT_ID(N'dbo.customer_history', N'U') IS NOT NULL
BEGIN
    INSERT INTO @results (object_name, rows_found)
    SELECT N'dbo.customer_history', COUNT_BIG(*)
    FROM dbo.customer_history
    WHERE customer_id = @cid
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END

SELECT object_name, rows_found
FROM @results;
```
