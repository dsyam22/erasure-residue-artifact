/* Read-only verification: is the erased subject's personal data still stored anywhere? */

DECLARE @cid int = 42;
DECLARE @name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname NOT NULL, rows_found bigint NOT NULL);

/* dbo.customer */
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

/* dbo.member (current rows) */
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.member', COUNT(*)
FROM dbo.member
WHERE member_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

/* dbo.member_history (temporal history) */
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.member_history', COUNT(*)
FROM dbo.member_history
WHERE member_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

/* CDC change table for dbo.customer */
IF EXISTS (SELECT 1 FROM sys.tables WHERE object_id = OBJECT_ID(N'cdc.dbo_customer_CT'))
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', COUNT(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @cid
       OR full_name = @name
       OR email = @email
       OR phone = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found) VALUES (N'cdc.dbo_customer_CT', 0);
END

/* Any other CDC capture instances in the database (dynamic) */
DECLARE @sql nvarchar(max) = N'';
DECLARE @obj sysname, @sch sysname, @tbl sysname;

DECLARE cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM cdc.change_tables ct
    JOIN sys.tables t ON t.object_id = ct.source_object_id
    JOIN sys.schemas s ON s.schema_id = t.schema_id
    WHERE NOT (s.name = N'dbo' AND t.name = N'customer');

OPEN cur;
FETCH NEXT FROM cur INTO @sch, @tbl;
WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj = @sch + N'.' + @tbl;
    SET @sql = @sql + N'
INSERT INTO #results (object_name, rows_found)
SELECT N''cdc.'' + ct.capture_instance + N''_CT'', COUNT(*)
FROM cdc.'' + QUOTENAME(ct.capture_instance) + N''_CT ct
CROSS APPLY (SELECT * FROM cdc.'' + QUOTENAME(ct.capture_instance) + N''_CT) x
WHERE 1=0;';
    FETCH NEXT FROM cur INTO @sch, @tbl;
END
CLOSE cur;
DEALLOCATE cur;

/* Simpler dynamic scan of every CDC change table for the subject values */
DECLARE @scan nvarchar(max) = N'';
SELECT @scan = @scan + N'
IF OBJECT_ID(N''cdc.' + QUOTENAME(ct.capture_instance) + N'_CT'') IS NOT NULL
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N''cdc.' + REPLACE(ct.capture_instance, '''', '''''') + N'_CT'', COUNT(*)
    FROM cdc.' + QUOTENAME(ct.capture_instance) + N'_CT
    WHERE (COL_LENGTH(N''cdc.' + QUOTENAME(ct.capture_instance) + N'_CT'', N''customer_id'') IS NOT NULL AND customer_id = @cid)
       OR (COL_LENGTH(N''cdc.' + QUOTENAME(ct.capture_instance) + N'_CT'', N''member_id'') IS NOT NULL AND member_id = @cid)
       OR (COL_LENGTH(N''cdc.' + QUOTENAME(ct.capture_instance) + N'_CT'', N''full_name'') IS NOT NULL AND full_name = @name)
       OR (COL_LENGTH(N''cdc.' + QUOTENAME(ct.capture_instance) + N'_CT'', N''email'') IS NOT NULL AND email = @email)
       OR (COL_LENGTH(N''cdc.' + QUOTENAME(ct.capture_instance) + N'_CT'', N''phone'') IS NOT NULL AND phone = @phone);
END'
FROM cdc.change_tables ct
WHERE ct.capture_instance NOT IN (N'dbo_customer');

IF @scan IS NOT NULL AND @scan <> N''
    EXEC sp_executesql @scan, N'@cid int, @name nvarchar(200), @email nvarchar(320), @phone nvarchar(50)',
        @cid = @cid, @name = @name, @email = @email, @phone = @phone;

/* Any other user tables that might contain the subject values (best-effort dynamic scan) */
DECLARE @other nvarchar(max) = N'';
SELECT @other = @other + N'
IF OBJECT_ID(N''' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) + N''') IS NOT NULL
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N''' + REPLACE(s.name + N'.' + t.name, '''', '''''') + N''', COUNT(*)
    FROM ' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) + N'
    WHERE (COL_LENGTH(N''' + REPLACE(s.name + N'.' + t.name, '''', '''''') + N''', N''customer_id'') IS NOT NULL AND customer_id = @cid)
       OR (COL_LENGTH(N''' + REPLACE(s.name + N'.' + t.name, '''', '''''') + N''', N''member_id'') IS NOT NULL AND member_id = @cid)
       OR (COL_LENGTH(N''' + REPLACE(s.name + N'.' + t.name, '''', '''''') + N''', N''full_name'') IS NOT NULL AND full_name = @name)
       OR (COL_LENGTH(N''' + REPLACE(s.name + N'.' + t.name, '''', '''''') + N''', N''email'') IS NOT NULL AND email = @email)
       OR (COL_LENGTH(N''' + REPLACE(s.name + N'.' + t.name, '''', '''''') + N''', N''phone'') IS NOT NULL AND phone = @phone);
END'
FROM sys.tables t
JOIN sys.schemas s ON s.schema_id = t.schema_id
WHERE t.is_ms_shipped = 0
  AND t.temporal_type <> 1  /* exclude history tables, handled explicitly */
  AND t.name NOT IN (N'customer', N'member')
  AND s.name NOT IN (N'cdc')
  AND t.name NOT LIKE N'%_CT'
  AND (EXISTS (SELECT 1 FROM sys.columns c WHERE c.object_id = t.object_id AND c.name IN (N'customer_id', N'member_id', N'full_name', N'email', N'phone')));

IF @other IS NOT NULL AND @other <> N''
    EXEC sp_executesql @other, N'@cid int, @name nvarchar(200), @email nvarchar(320), @phone nvarchar(50)',
        @cid = @cid, @name = @name, @email = @email, @phone = @phone;

SELECT object_name, rows_found FROM #results ORDER BY object_name;

DROP TABLE #results;
