FROM (

           (SELECT COUNT(*) FROM dbo.customer

    UNION ALL
    SELECT 'dbo.member',
           (SELECT COUNT(*) FROM dbo.member
            WHERE member_id = 42
              AND (full_name = N'Subject_0000042'
                   OR email = N'subject42@example.invalid'
                   OR phone = N'+1-555-0000042'))
    UNION ALL
    SELECT 'dbo.member_history',
           (SELECT COUNT(*) FROM dbo.member_history
            WHERE member_id = 42
              AND (full_name = N'Subject_0000042'
                   OR email = N'subject42@example.invalid'
                   OR phone = N'+1-555-0000042'))
    UNION ALL
    SELECT 'cdc.dbo_customer_CT',
           (SELECT COUNT(*) FROM cdc.dbo_customer_CT
            WHERE customer_id = 42
              AND (full_name = N'Subject_0000042'
                   OR email = N'subject42@example.invalid'
                   OR phone = N'+1-555-0000042'))
    UNION ALL
    SELECT 'cdc.dbo_member_CT',
           (SELECT COUNT(*) FROM cdc.dbo_member_CT
            WHERE member_id = 42
              AND (full_name = N'Subject_0000042'
                   OR email = N'subject42@example.invalid'
                   OR phone = N'+1-555-0000042'))
    UNION ALL
    SELECT 'cdc.dbo_member_history_CT',

              AND (full_name = N'Subject_0000042'
