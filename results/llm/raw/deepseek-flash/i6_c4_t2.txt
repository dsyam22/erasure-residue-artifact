```sql
SELECT object_name, rows_found
FROM (
    SELECT 'cdc.dbo_customer_CT' AS object_name,
           COUNT(*) AS rows_found
    FROM cdc.dbo_customer_CT
    WHERE customer_id = 42
      AND phone = '+1-555-0000042'
) AS a
UNION ALL
SELECT 'cdc.ddl_history' AS object_name,
       COUNT(*) AS rows_found
FROM cdc.ddl_history
WHERE ddl_command LIKE '%+1-555-0000042%'
UNION ALL
SELECT 'cdc.captured_columns' AS object_name,
       COUNT(*) AS rows_found
FROM cdc.captured_columns
WHERE column_name LIKE '%+1-555-0000042%'
UNION ALL
SELECT 'cdc.change_tables' AS object_name,
       COUNT(*) AS rows_found
FROM cdc.change_tables
WHERE capture_instance LIKE '%+1-555-0000042%'
UNION ALL
SELECT 'cdc.index_columns' AS object_name,
       COUNT(*) AS rows_found
FROM cdc.index_columns
WHERE column_name LIKE '%+1-555-0000042%'
UNION ALL
SELECT 'cdc.lsn_time_mapping' AS object_name,
       COUNT(*) AS rows_found
FROM cdc.lsn_time_mapping
WHERE CAST(tran_id AS nvarchar(max)) LIKE '%+1-555-0000042%'
UNION ALL
SELECT 'dbo.systranschemas' AS object_name,
       COUNT(*) AS rows_found
FROM dbo.systranschemas
WHERE CAST(tabid AS nvarchar(max)) LIKE '%+1-555-0000042%'
   OR CAST(startlsn AS nvarchar(max)) LIKE '%+1-555-0000042%'
   OR CAST(endlsn AS nvarchar(max)) LIKE '%+1-555-0000042%'
   OR CAST(typeid AS nvarchar(max)) LIKE '%+1-555-0000042%'
UNION ALL
SELECT 'dbo.customer' AS object_name,
       COUNT(*) AS rows_found
FROM dbo.customer
WHERE customer_id = 42
  AND (full_name LIKE '%+1-555-0000042%' OR email LIKE '%+1-555-0000042%');
```
