```sql
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @full_name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname NOT NULL, rows_found bigint NOT NULL);

/* dbo.customer: search all text columns for any of the subject's identifiers */
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT_BIG(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* temporal history table, if it exists */
IF EXISTS (SELECT 1 FROM sys.tables WHERE object_id = OBJECT_ID(N'dbo.customer') AND temporal_type = 1)
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'dbo.customer_history', COUNT_BIG(*)
    FROM dbo.customer_history
    WHERE customer_id = @cid
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'dbo.customer_history', 0;
END;

/* CDC change tables, if CDC is enabled on the database */
IF EXISTS (SELECT 1 FROM sys.databases WHERE name = DB_NAME() AND is_cdc_enabled = 1)
BEGIN
    IF OBJECT_ID(N'cdc.dbo_customer_CT') IS NOT NULL
    BEGIN
        INSERT INTO #results (object_name, rows_found)
        SELECT N'cdc.dbo_customer_CT', COUNT_BIG(*)
        FROM cdc.dbo_customer_CT
        WHERE customer_id = @cid
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone;
    END
    ELSE
    BEGIN
        INSERT INTO #results (object_name, rows_found)
        SELECT N'cdc.dbo_customer_CT', 0;
    END;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', 0;
END;

SELECT object_name, rows_found
FROM #results;

DROP TABLE #results;
```
