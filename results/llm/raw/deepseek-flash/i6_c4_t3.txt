```sql
SELECT object_name, rows_found
FROM (
    SELECT 'dbo.customer' AS object_name,
           (SELECT COUNT(*) FROM dbo.customer
            WHERE CONVERT(nvarchar(max), full_name) LIKE N'%Subject_0000042%'
               OR CONVERT(nvarchar(max), email) LIKE N'%subject42@example.invalid%'
               OR CONVERT(nvarchar(max), email) LIKE N'%Subject_0000042%') AS rows_found
    UNION ALL
    SELECT 'cdc.dbo_customer_CT' AS object_name,
           (SELECT COUNT(*) FROM cdc.dbo_customer_CT
            WHERE CONVERT(nvarchar(max), full_name) LIKE N'%Subject_0000042%'
               OR CONVERT(nvarchar(max), email) LIKE N'%subject42@example.invalid%'
               OR CONVERT(nvarchar(max), email) LIKE N'%Subject_0000042%'
               OR CONVERT(nvarchar(max), phone) LIKE N'%+1-555-0000042%'
               OR CONVERT(nvarchar(max), phone) LIKE N'%5550000042%') AS rows_found
    UNION ALL
    SELECT 'cdc.captured_columns' AS object_name,
           (SELECT COUNT(*) FROM cdc.captured_columns
            WHERE CONVERT(nvarchar(max), column_name) LIKE N'%phone%') AS rows_found
    UNION ALL
    SELECT 'cdc.ddl_history' AS object_name,
           (SELECT COUNT(*) FROM cdc.ddl_history
            WHERE CONVERT(nvarchar(max), ddl_command) LIKE N'%phone%'
               OR CONVERT(nvarchar(max), ddl_command) LIKE N'%Subject_0000042%'
               OR CONVERT(nvarchar(max), ddl_command) LIKE N'%subject42@example.invalid%'
               OR CONVERT(nvarchar(max), ddl_command) LIKE N'%5550000042%') AS rows_found
    UNION ALL
    SELECT 'dbo.systranschemas' AS object_name,
           (SELECT COUNT(*) FROM dbo.systranschemas
            WHERE CONVERT(nvarchar(max), typeid) LIKE N'%Subject_0000042%') AS rows_found
) AS checks
ORDER BY object_name;
```
