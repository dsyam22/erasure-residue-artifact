```sql
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @fn nvarchar(200) = N'Subject_0000042';
DECLARE @em nvarchar(400) = N'subject42@example.invalid';
DECLARE @ph nvarchar(100) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname, rows_found int);

INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT_BIG(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @fn
   OR email = @em
   OR phone = @ph;

INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer_history', COUNT_BIG(*)
FROM dbo.customer_history
WHERE customer_id = @cid
   OR full_name = @fn
   OR email = @em
   OR phone = @ph;

SELECT object_name, rows_found
FROM #results;

DROP TABLE #results;
```
