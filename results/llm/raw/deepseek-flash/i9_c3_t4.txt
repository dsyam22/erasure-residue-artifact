/* Read-only verification of residual personal data for customer_id 42.
   Checks the live table, the system-versioned temporal history table,
   and CDC change tables if CDC capture instances exist.
   Returns exactly one result set: object_name, rows_found. */

SET NOCOUNT ON;

DECLARE @cid   int           = 42;
DECLARE @name  nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50)  = N'+1-555-0000042';

CREATE TABLE #res
(
    object_name sysname      NOT NULL,
    rows_found  bigint       NOT NULL,
    sort_order  int          NOT NULL
);

/* 1. Live table dbo.customer */
INSERT INTO #res (object_name, rows_found, sort_order)
SELECT N'dbo.customer',
       COUNT_BIG(*),
       1
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name   = @name
   OR email       = @email
   OR phone       = @phone;

/* 2. Temporal history table dbo.customer_history */
INSERT INTO #res (object_name, rows_found, sort_order)
SELECT N'dbo.customer_history',
       COUNT_BIG(*),
       2
FROM dbo.customer_history
WHERE customer_id = @cid
   OR full_name   = @name
   OR email       = @email
   OR phone       = @phone;

/* 3. CDC change tables (only if CDC capture instances exist for dbo.customer).
      Built dynamically because change-table names are generated by SQL Server. */
DECLARE @sql nvarchar(max) = N'';

SELECT @sql = @sql + N'
INSERT INTO #res (object_name, rows_found, sort_order)
SELECT N''' + QUOTENAME(OBJECT_SCHEMA_NAME(ct.object_id)) + N'.'
             + QUOTENAME(OBJECT_NAME(ct.object_id)) + N''',
       COUNT_BIG(*),
       3
FROM ' + QUOTENAME(OBJECT_SCHEMA_NAME(ct.object_id)) + N'.'
        + QUOTENAME(OBJECT_NAME(ct.object_id)) + N' AS ct
WHERE ct.customer_id = @cid
   OR ct.full_name   = @name
   OR ct.email       = @email
   OR ct.phone       = @phone;
'
FROM cdc.change_tables AS ct
WHERE ct.source_object_id = OBJECT_ID(N'dbo.customer');

IF @sql <> N''
BEGIN
    EXEC sp_executesql
        @sql,
        N'@cid int, @name nvarchar(200), @email nvarchar(320), @phone nvarchar(50)',
        @cid = @cid, @name = @name, @email = @email, @phone = @phone;
END

/* Exactly one result set. */
SELECT object_name, rows_found
FROM #res
ORDER BY sort_order, object_name;

DROP TABLE #res;
