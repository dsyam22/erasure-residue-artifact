```sql
/* Read-only verification of residual personal data for subject 42 */
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @full_name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(200) = N'subject42@example.invalid';
DECLARE @phone nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname NOT NULL, rows_found bigint NOT NULL);

/* 1) Current table: dbo.customer */
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT_BIG(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* 2) Temporal history table: dbo.customer_history */
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer_history', COUNT_BIG(*)
FROM dbo.customer_history
WHERE customer_id = @cid
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

/* 3) CDC change tables: cdc.dbo_customer_CT (if CDC is enabled) */
IF OBJECT_ID(N'cdc.dbo_customer_CT', N'U') IS NOT NULL
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', COUNT_BIG(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @cid
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', 0;
END;

/* 4) Change tracking side table: sys.change_tracking_tables (metadata only, no personal data) */
INSERT INTO #results (object_name, rows_found)
SELECT N'sys.change_tracking_tables', 0;

/* Return exactly one result set */
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
