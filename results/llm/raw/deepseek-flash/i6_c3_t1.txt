```sql
/* Read-only verification of residual erased personal data (phone only) */

DECLARE @subject_id int = 42;
DECLARE @phone nvarchar(100) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname, rows_found bigint);

/* Current customer table (phone column should be gone) */
IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('dbo.customer') AND name = 'phone')
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'dbo.customer', COUNT(*) FROM dbo.customer WHERE phone = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found) VALUES ('dbo.customer', 0);
END

/* CDC change table - capture instance captured_columns includes phone */
IF OBJECT_ID('cdc.dbo_customer_CT') IS NOT NULL
BEGIN
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('cdc.dbo_customer_CT') AND name = 'phone')
    BEGIN
        INSERT INTO #results (object_name, rows_found)
        SELECT 'cdc.dbo_customer_CT', COUNT(*)
        FROM cdc.dbo_customer_CT
        WHERE phone = @phone
          AND customer_id = @subject_id;
    END
    ELSE
    BEGIN
        INSERT INTO #results (object_name, rows_found) VALUES ('cdc.dbo_customer_CT', 0);
    END
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found) VALUES ('cdc.dbo_customer_CT', 0);
END

/* Check any other CDC capture instances / change tables that might reference the phone */
DECLARE @ct sysname, @sql nvarchar(max);
DECLARE ct_cursor CURSOR LOCAL FAST_FORWARD FOR
    SELECT QUOTENAME(OBJECT_SCHEMA_NAME(object_id)) + '.' + QUOTENAME(name)
    FROM sys.tables
    WHERE name LIKE '%_CT'
      AND OBJECT_SCHEMA_NAME(object_id) = 'cdc'
      AND name <> 'dbo_customer_CT';

OPEN ct_cursor;
FETCH NEXT FROM ct_cursor INTO @ct;
WHILE @@FETCH_STATUS = 0
BEGIN
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@ct) AND name = 'phone')
    BEGIN
        SET @sql = N'INSERT INTO #results (object_name, rows_found)
                     SELECT @obj, COUNT(*) FROM ' + @ct + N' WHERE phone = @ph';
        EXEC sp_executesql @sql, N'@obj sysname, @ph nvarchar(100)', @obj = @ct, @ph = @phone;
    END
    ELSE
    BEGIN
        INSERT INTO #results (object_name, rows_found) VALUES (@ct, 0);
    END
    FETCH NEXT FROM ct_cursor INTO @ct;
END
CLOSE ct_cursor;
DEALLOCATE ct_cursor;

/* Temporal history tables on dbo.customer (none expected, but check) */
IF OBJECT_ID('dbo.customer_history') IS NOT NULL
BEGIN
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('dbo.customer_history') AND name = 'phone')
    BEGIN
        INSERT INTO #results (object_name, rows_found)
        SELECT 'dbo.customer_history', COUNT(*) FROM dbo.customer_history WHERE phone = @phone;
    END
    ELSE
    BEGIN
        INSERT INTO #results (object_name, rows_found) VALUES ('dbo.customer_history', 0);
    END
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found) VALUES ('dbo.customer_history', 0);
END

/* Any other table in the database that still has a phone column */
DECLARE @tbl sysname;
DECLARE tbl_cursor CURSOR LOCAL FAST_FORWARD FOR
    SELECT QUOTENAME(OBJECT_SCHEMA_NAME(c.object_id)) + '.' + QUOTENAME(OBJECT_NAME(c.object_id))
    FROM sys.columns c
    JOIN sys.tables t ON t.object_id = c.object_id
    WHERE c.name = 'phone'
      AND t.is_ms_shipped = 0
      AND OBJECT_NAME(c.object_id) <> 'customer'
      AND OBJECT_SCHEMA_NAME(c.object_id) <> 'cdc'
      AND OBJECT_NAME(c.object_id) NOT LIKE '%_history'
      AND OBJECT_NAME(c.object_id) NOT LIKE '%_CT';

OPEN tbl_cursor;
FETCH NEXT FROM tbl_cursor INTO @tbl;
WHILE @@FETCH_STATUS = 0
BEGIN
    SET @sql = N'INSERT INTO #results (object_name, rows_found)
                 SELECT @obj, COUNT(*) FROM ' + @tbl + N' WHERE phone = @ph';
    EXEC sp_executesql @sql, N'@obj sysname, @ph nvarchar(100)', @obj = @tbl, @ph = @phone;
    FETCH NEXT FROM tbl_cursor INTO @tbl;
END
CLOSE tbl_cursor;
DEALLOCATE tbl_cursor;

SELECT object_name, rows_found FROM #results ORDER BY object_name;

DROP TABLE #results;
```
