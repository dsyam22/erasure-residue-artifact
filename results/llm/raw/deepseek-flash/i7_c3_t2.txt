/* Read-only verification: does the erased subject's personal data still exist anywhere? */

DECLARE @cid   int           = 42;
DECLARE @name  nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(200) = N'subject42@example.invalid';
DECLARE @phone nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname, rows_found bigint);

/* 1. Live base table */
INSERT INTO #results (object_name, rows_found)
SELECT 'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name   = @name
   OR email       = @email
   OR phone       = @phone;

/* 2. Downstream replica */
INSERT INTO #results (object_name, rows_found)
SELECT 'reporting.customer_replica', COUNT(*)
FROM reporting.customer_replica
WHERE customer_id = @cid
   OR full_name   = @name
   OR email       = @email
   OR phone       = @phone;

/* 3. CDC change tables (if CDC enabled on dbo.customer) */
IF EXISTS (SELECT 1 FROM sys.tables WHERE name = 'customer_CT' AND schema_id = SCHEMA_ID('cdc'))
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'cdc.dbo_customer_CT', COUNT(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @cid
       OR full_name   = @name
       OR email       = @email
       OR phone       = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'cdc.dbo_customer_CT', 0;
END

/* 4. Temporal history table (if dbo.customer is system-versioned) */
IF EXISTS (SELECT 1 FROM sys.tables WHERE name = 'customer_history' AND schema_id = SCHEMA_ID('dbo'))
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'dbo.customer_history', COUNT(*)
    FROM dbo.customer_history
    WHERE customer_id = @cid
       OR full_name   = @name
       OR email       = @email
       OR phone       = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'dbo.customer_history', 0;
END

/* 5. Temporal history table for the replica (if system-versioned) */
IF EXISTS (SELECT 1 FROM sys.tables WHERE name = 'customer_replica_history' AND schema_id = SCHEMA_ID('reporting'))
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'reporting.customer_replica_history', COUNT(*)
    FROM reporting.customer_replica_history
    WHERE customer_id = @cid
       OR full_name   = @name
       OR email       = @email
       OR phone       = @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'reporting.customer_replica_history', 0;
END

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
