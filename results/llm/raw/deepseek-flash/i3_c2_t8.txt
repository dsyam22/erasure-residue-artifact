/* Read-only verification: is the erased subject's personal data still stored anywhere? */
DECLARE @cid int = 42;
DECLARE @name nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(200) = N'subject42@example.invalid';
DECLARE @phone nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results (object_name sysname, rows_found bigint);

/* 1. dbo.customer - live table */
INSERT INTO #results
SELECT 'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

/* 2. dbo.member - live temporal table (current rows) */
INSERT INTO #results
SELECT 'dbo.member', COUNT(*)
FROM dbo.member
WHERE member_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

/* 3. dbo.member_history - temporal history table */
INSERT INTO #results
SELECT 'dbo.member_history', COUNT(*)
FROM dbo.member_history
WHERE member_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

/* 4. cdc.dbo_customer_CT - CDC change table */
INSERT INTO #results
SELECT 'cdc.dbo_customer_CT', COUNT(*)
FROM cdc.dbo_customer_CT
WHERE customer_id = @cid
   OR full_name = @name
   OR email = @email
   OR phone = @phone;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;
