/* ============================================================================
   Verification script: is the erased subject's original personal data still
   present anywhere in this database?
   Read-only. Returns exactly one result set: object_name, rows_found.
   ============================================================================ */

SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name  nvarchar(4000) = N'Subject_0000042';
DECLARE @email      nvarchar(4000) = N'subject42@example.invalid';
DECLARE @phone      nvarchar(4000) = N'+1-555-0000042';

/* ---------------------------------------------------------------------------
   Result accumulation
   --------------------------------------------------------------------------- */
IF OBJECT_ID('tempdb..#results') IS NOT NULL DROP TABLE #results;
CREATE TABLE #results (object_name sysname NOT NULL, rows_found bigint NOT NULL);

/* ---------------------------------------------------------------------------
   1) dbo.customer  (live table)
   --------------------------------------------------------------------------- */
INSERT INTO #results (object_name, rows_found)
SELECT 'dbo.customer',
       COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
  AND (
        full_name = @full_name
     OR email     = @email
     OR phone     = @phone
      );

/* ---------------------------------------------------------------------------
   2) dbo.member  (live table, current row)
   --------------------------------------------------------------------------- */
INSERT INTO #results (object_name, rows_found)
SELECT 'dbo.member',
       COUNT(*)
FROM dbo.member
WHERE member_id = @member_id
  AND (
        full_name = @full_name
     OR email     = @email
     OR phone     = @phone
      );

/* ---------------------------------------------------------------------------
   3) dbo.member_history  (system-versioned temporal history table)
   --------------------------------------------------------------------------- */
INSERT INTO #results (object_name, rows_found)
SELECT 'dbo.member_history',
       COUNT(*)
FROM dbo.member_history
WHERE member_id = @member_id
  AND (
        full_name = @full_name
     OR email     = @email
     OR phone     = @phone
      );

/* ---------------------------------------------------------------------------
   4) cdc.dbo_customer_CT  (CDC change table)
   --------------------------------------------------------------------------- */
IF OBJECT_ID('cdc.dbo_customer_CT') IS NOT NULL
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'cdc.dbo_customer_CT',
           COUNT(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @customer_id
      AND (
            full_name = @full_name
         OR email     = @email
         OR phone     = @phone
          );
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT 'cdc.dbo_customer_CT', 0;
END

/* ---------------------------------------------------------------------------
   Return the single result set
   --------------------------------------------------------------------------- */
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
