```sql
-- SCRIPT
SET NOCOUNT ON;

DECLARE @cid int = 42;
DECLARE @full nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

DECLARE @results TABLE (object_name sysname, rows_found int);

-- Check dbo.customer for any residual rows matching the subject's original personal data
INSERT INTO @results (object_name, rows_found)
SELECT 'dbo.customer',
       COUNT(*)
FROM dbo.customer
WHERE customer_id = @cid
   OR full_name = @full
   OR email = @email
   OR phone = @phone;

SELECT object_name, rows_found
FROM @results
ORDER BY object_name;
```
