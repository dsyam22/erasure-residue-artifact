DECLARE @VerificationResults TABLE (

    rows_found INT
);

    customer_id = 42

INSERT INTO @VerificationResults (object_name, rows_found)

    'reporting.customer_replica', 
    COUNT(*)
FROM 
    reporting.customer_replica
WHERE 
    customer_id = 42 
    OR full_name = 'Subject_0000042' 
    OR email = 'subject42@example.invalid'
