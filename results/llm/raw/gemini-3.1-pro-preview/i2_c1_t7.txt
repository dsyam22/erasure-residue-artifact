```sql
SET NOCOUNT ON;

DECLARE @Results TABLE (
    object_name NVARCHAR(512),
    rows_found INT
);

DECLARE @count INT;

-- Check the main table
SELECT @count = COUNT(*) 
FROM dbo.customer 
WHERE customer_id = 42 
   OR full_name = 'Subject_0000042' 
   OR email = 'subject42@example.invalid' 
   OR phone = '+1-555-0000042';

INSERT INTO @Results (object_name, rows_found)
VALUES ('dbo.customer', @count);

-- Check the history table if dbo.customer is a system-versioned temporal table
DECLARE @HistoryTableSchema NVARCHAR(128);
DECLARE @HistoryTableName NVARCHAR(128);
DECLARE @Sql NVARCHAR(MAX);

SELECT 
    @HistoryTableSchema = SCHEMA_NAME(h.schema_id),
    @HistoryTableName = h.name
FROM sys.tables t
JOIN sys.tables h ON t.history_table_id = h.object_id
WHERE t.object_id = OBJECT_ID('dbo.customer');

IF @HistoryTableName IS NOT NULL
BEGIN
    SET @Sql = N'SELECT @cnt = COUNT(*) FROM ' + QUOTENAME(@HistoryTableSchema) + N'.' + QUOTENAME(@HistoryTableName) + 
               N' WHERE customer_id = 42 OR full_name = ''Subject_0000042'' OR email = ''subject42@example.invalid'' OR phone = ''+1-555-0000042'';';
    
    EXEC sp_executesql @Sql, N'@cnt INT OUTPUT', @cnt = @count OUTPUT;
    
    INSERT INTO @Results (object_name, rows_found)
    VALUES (@HistoryTableSchema + '.' + @HistoryTableName, @count);
END

-- Return the final result set
SELECT object_name, rows_found 
FROM @Results;
```
