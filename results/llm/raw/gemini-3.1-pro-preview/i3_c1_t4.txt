```sql
SET NOCOUNT ON;

CREATE TABLE #Results (
    object_name NVARCHAR(255),
    rows_found INT
);

-- Check dbo.customer
INSERT INTO #Results (object_name, rows_found)
SELECT 'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = 42 
   OR full_name = 'Subject_0000042' 
   OR email = 'subject42@example.invalid' 
   OR phone = '+1-555-0000042';

-- Check dbo.member (dynamically checking if it is a system-versioned temporal table)
DECLARE @sql NVARCHAR(MAX);

IF EXISTS (SELECT 1 FROM sys.tables WHERE object_id = OBJECT_ID('dbo.member') AND temporal_type = 2)
BEGIN
    SET @sql = N'
    INSERT INTO #Results (object_name, rows_found)
    SELECT ''dbo.member (including temporal history)'', COUNT(*)
    FROM dbo.member FOR SYSTEM_TIME ALL
    WHERE member_id = 42 
       OR full_name = ''Subject_0000042'' 
       OR email = ''subject42@example.invalid'' 
       OR phone = ''+1-555-0000042'';
    ';
    EXEC sp_executesql @sql;
END
ELSE
BEGIN
    INSERT INTO #Results (object_name, rows_found)
    SELECT 'dbo.member', COUNT(*)
    FROM dbo.member
    WHERE member_id = 42 
       OR full_name = 'Subject_0000042' 
       OR email = 'subject42@example.invalid' 
       OR phone = '+1-555-0000042';
END

-- Return the final result set
SELECT object_name, rows_found 
FROM #Results;

DROP TABLE #Results;
```
