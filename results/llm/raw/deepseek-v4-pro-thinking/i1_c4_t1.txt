SELECT 'dbo.customer' AS object_name,
       COUNT(*) AS rows_found
FROM dbo.customer
WHERE customer_id = 42
  AND full_name = N'Subject_0000042'
  AND email = N'subject42@example.invalid'
  AND phone = N'+1-555-0000042'
UNION ALL
SELECT 'cdc.dbo_customer_CT',
       COUNT(*)
FROM cdc.dbo_customer_CT
WHERE customer_id = 42
  AND full_name = N'Subject_0000042'
  AND email = N'subject42@example.invalid'
  AND phone = N'+1-555-0000042';
