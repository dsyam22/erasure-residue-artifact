```sql
SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

SELECT
    o.object_name,
    SUM(CASE WHEN r.rows_found IS NULL THEN 0 ELSE r.rows_found END) AS rows_found
FROM
    (
        SELECT N'dbo.customer' AS object_name
        UNION ALL SELECT N'reporting.customer_replica'
    ) o
LEFT JOIN
    (
        SELECT N'dbo.customer' AS object_name,
               COUNT(*) AS rows_found
        FROM dbo.customer
        WHERE customer_id = @customer_id
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone

        UNION ALL

        SELECT N'reporting.customer_replica' AS object_name,
               COUNT(*) AS rows_found
        FROM reporting.customer_replica
        WHERE customer_id = @customer_id
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone
    ) r ON o.object_name = r.object_name
GROUP BY o.object_name
ORDER BY o.object_name;
```
