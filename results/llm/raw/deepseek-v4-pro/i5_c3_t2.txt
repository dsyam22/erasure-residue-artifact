```sql
DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(255),
    rows_found int
);

-- Check current table
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer',
    COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check temporal history table
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer_history',
    COUNT(*)
FROM dbo.customer_history
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC change tables (if any exist)
IF EXISTS (SELECT 1 FROM sys.tables WHERE name LIKE 'cdc.%' AND is_tracked_by_cdc = 1)
BEGIN
    DECLARE @cdc_sql nvarchar(max) = N'';
    SELECT @cdc_sql = @cdc_sql + 
        N'INSERT INTO #results (object_name, rows_found) SELECT ''' + 
        QUOTENAME(SCHEMA_NAME(schema_id)) + N'.' + QUOTENAME(name) + N''', COUNT(*) FROM ' + 
        QUOTENAME(SCHEMA_NAME(schema_id)) + N'.' + QUOTENAME(name) + 
        N' WHERE customer_id = @customer_id OR full_name = @full_name OR email = @email OR phone = @phone; '
    FROM sys.tables
    WHERE name LIKE 'cdc.%' AND is_tracked_by_cdc = 1;
    
    IF @cdc_sql <> N''
    BEGIN
        EXEC sp_executesql @cdc_sql, 
            N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255)',
            @customer_id, @full_name, @email, @phone;
    END
END

-- Check any other user tables that might contain copies (dynamic search)
DECLARE @table_name nvarchar(255);
DECLARE @schema_name nvarchar(255);
DECLARE @full_table_name nvarchar(510);
DECLARE @sql nvarchar(max);

DECLARE table_cursor CURSOR FOR
SELECT 
    s.name AS schema_name,
    t.name AS table_name
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE t.name NOT IN ('customer', 'customer_history')
  AND t.name NOT LIKE 'cdc.%'
  AND t.is_ms_shipped = 0;

OPEN table_cursor;
FETCH NEXT FROM table_cursor INTO @schema_name, @table_name;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @full_table_name = QUOTENAME(@schema_name) + N'.' + QUOTENAME(@table_name);
    
    -- Check if table has any of the potential columns
    IF EXISTS (
        SELECT 1 
        FROM sys.columns c 
        WHERE c.object_id = OBJECT_ID(@full_table_name) 
          AND c.name IN ('customer_id', 'full_name', 'email', 'phone')
    )
    BEGIN
        SET @sql = N'INSERT INTO #results (object_name, rows_found) SELECT ''' + @full_table_name + N''', COUNT(*) FROM ' + @full_table_name + N' WHERE 1=0';
        
        IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'customer_id')
            SET @sql = @sql + N' OR customer_id = @customer_id';
        IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'full_name')
            SET @sql = @sql + N' OR full_name = @full_name';
        IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'email')
            SET @sql = @sql + N' OR email = @email';
        IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'phone')
            SET @sql = @sql + N' OR phone = @phone';
        
        SET @sql = REPLACE(@sql, 'WHERE 1=0 OR', 'WHERE');
        
        EXEC sp_executesql @sql, 
            N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255)',
            @customer_id, @full_name, @email, @phone;
    END
    
    FETCH NEXT FROM table_cursor INTO @schema_name, @table_name;
END

CLOSE table_cursor;
DEALLOCATE table_cursor;

-- Return results
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
