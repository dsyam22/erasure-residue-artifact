```sql
DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(255),
    rows_found bigint
);

-- Check current table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC change table
INSERT INTO #results (object_name, rows_found)
SELECT N'cdc.dbo_customer_CT', COUNT(*)
FROM cdc.dbo_customer_CT
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check for temporal history tables (none in this configuration, but check if exists)
IF EXISTS (SELECT 1 FROM sys.tables WHERE name = 'customer' AND temporal_type = 2)
BEGIN
    DECLARE @history_table nvarchar(255);
    SELECT @history_table = QUOTENAME(SCHEMA_NAME(schema_id)) + N'.' + QUOTENAME(history_table_name)
    FROM sys.tables
    WHERE name = 'customer' AND temporal_type = 2;
    
    IF @history_table IS NOT NULL
    BEGIN
        DECLARE @sql nvarchar(max) = N'SELECT @cnt = COUNT(*) FROM ' + @history_table + 
            N' WHERE customer_id = @cid OR full_name = @fn OR email = @em OR phone = @ph';
        DECLARE @cnt bigint;
        EXEC sp_executesql @sql, N'@cid int, @fn nvarchar(255), @em nvarchar(255), @ph nvarchar(255), @cnt bigint OUTPUT',
            @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone, @cnt = @cnt OUTPUT;
        
        INSERT INTO #results (object_name, rows_found)
        SELECT @history_table, @cnt;
    END
END

-- Check for any other tables that might contain copies (common patterns)
-- Check for tables with similar column names
DECLARE @table_name nvarchar(255);
DECLARE @schema_name nvarchar(255);
DECLARE @full_table_name nvarchar(255);

DECLARE table_cursor CURSOR FOR
SELECT s.name, t.name
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE t.name != 'customer'
  AND EXISTS (
    SELECT 1 FROM sys.columns c 
    WHERE c.object_id = t.object_id 
      AND (c.name IN ('customer_id', 'full_name', 'email', 'phone'))
  );

OPEN table_cursor;
FETCH NEXT FROM table_cursor INTO @schema_name, @table_name;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @full_table_name = QUOTENAME(@schema_name) + N'.' + QUOTENAME(@table_name);
    
    DECLARE @check_sql nvarchar(max) = N'SELECT @cnt = COUNT(*) FROM ' + @full_table_name + 
        N' WHERE 1=0';
    
    -- Build dynamic WHERE clause based on existing columns
    DECLARE @where_clause nvarchar(max) = N'';
    
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'customer_id')
        SET @where_clause = @where_clause + N' OR customer_id = @cid';
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'full_name')
        SET @where_clause = @where_clause + N' OR full_name = @fn';
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'email')
        SET @where_clause = @where_clause + N' OR email = @em';
    IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID(@full_table_name) AND name = 'phone')
        SET @where_clause = @where_clause + N' OR phone = @ph';
    
    IF LEN(@where_clause) > 0
    BEGIN
        SET @where_clause = STUFF(@where_clause, 1, 4, N' WHERE ');
        SET @check_sql = N'SELECT @cnt = COUNT(*) FROM ' + @full_table_name + @where_clause;
        
        SET @cnt = 0;
        EXEC sp_executesql @check_sql, N'@cid int, @fn nvarchar(255), @em nvarchar(255), @ph nvarchar(255), @cnt bigint OUTPUT',
            @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone, @cnt = @cnt OUTPUT;
        
        INSERT INTO #results (object_name, rows_found)
        SELECT @full_table_name, @cnt;
    END
    
    FETCH NEXT FROM table_cursor INTO @schema_name, @table_name;
END

CLOSE table_cursor;
DEALLOCATE table_cursor;

-- Return results
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
