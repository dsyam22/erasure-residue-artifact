```sql
DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(255),
    rows_found int
);

-- Check current table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check temporal history table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer_history', COUNT(*)
FROM dbo.customer_history
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check for CDC change tables (if CDC was ever enabled, but config says cdc_enabled=0, still check if tables exist)
IF EXISTS (SELECT 1 FROM sys.tables WHERE name = 'customer_CT' AND schema_id = SCHEMA_ID('cdc'))
BEGIN
    DECLARE @cdc_sql nvarchar(max) = N'
        SELECT N''cdc.customer_CT'', COUNT(*)
        FROM cdc.customer_CT
        WHERE customer_id = @customer_id
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone';
    INSERT INTO #results (object_name, rows_found)
    EXEC sp_executesql @cdc_sql, N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255)',
                       @customer_id, @full_name, @email, @phone;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.customer_CT', 0;
END

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
