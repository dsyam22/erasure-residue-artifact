```sql
SET NOCOUNT ON;

DECLARE @customer_id INT = 42;
DECLARE @full_name NVARCHAR(255) = N'Subject_0000042';
DECLARE @email NVARCHAR(255) = N'subject42@example.invalid';
DECLARE @phone NVARCHAR(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name NVARCHAR(255),
    rows_found INT
);

-- Check dbo.customer
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer',
    COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check dbo.customer_history
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer_history',
    COUNT(*)
FROM dbo.customer_history
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC change tables (if any exist for dbo.customer)
IF EXISTS (
    SELECT 1 
    FROM sys.objects 
    WHERE name LIKE 'cdc.%_customer%' 
      AND type = 'U'
)
BEGIN
    DECLARE @cdc_table NVARCHAR(255);
    DECLARE cdc_cursor CURSOR FOR
        SELECT QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
        FROM sys.tables t
        JOIN sys.schemas s ON t.schema_id = s.schema_id
        WHERE t.name LIKE 'cdc.%_customer%'
          AND t.is_ms_shipped = 0;

    OPEN cdc_cursor;
    FETCH NEXT FROM cdc_cursor INTO @cdc_table;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        DECLARE @sql NVARCHAR(MAX);
        SET @sql = N'
            INSERT INTO #results (object_name, rows_found)
            SELECT 
                N''' + @cdc_table + N''',
                COUNT(*)
            FROM ' + @cdc_table + N'
            WHERE customer_id = @customer_id
               OR full_name = @full_name
               OR email = @email
               OR phone = @phone';
        
        EXEC sp_executesql @sql, 
            N'@customer_id INT, @full_name NVARCHAR(255), @email NVARCHAR(255), @phone NVARCHAR(255)',
            @customer_id, @full_name, @email, @phone;

        FETCH NEXT FROM cdc_cursor INTO @cdc_table;
    END;

    CLOSE cdc_cursor;
    DEALLOCATE cdc_cursor;
END

-- Check if there are any other tables that might contain a copy of the data
-- Look for tables with similar column names
IF EXISTS (
    SELECT 1 
    FROM sys.tables t
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    JOIN sys.columns c ON t.object_id = c.object_id
    WHERE c.name IN ('customer_id', 'full_name', 'email', 'phone')
      AND t.name NOT IN ('customer', 'customer_history')
      AND t.is_ms_shipped = 0
)
BEGIN
    DECLARE @other_table NVARCHAR(255);
    DECLARE other_cursor CURSOR FOR
        SELECT DISTINCT QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
        FROM sys.tables t
        JOIN sys.schemas s ON t.schema_id = s.schema_id
        JOIN sys.columns c ON t.object_id = c.object_id
        WHERE c.name IN ('customer_id', 'full_name', 'email', 'phone')
          AND t.name NOT IN ('customer', 'customer_history')
          AND t.is_ms_shipped = 0;

    OPEN other_cursor;
    FETCH NEXT FROM other_cursor INTO @other_table;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        -- Check if this table has all the relevant columns
        DECLARE @has_cols INT;
        SELECT @has_cols = COUNT(*)
        FROM sys.columns c
        JOIN sys.tables t ON c.object_id = t.object_id
        JOIN sys.schemas s ON t.schema_id = s.schema_id
        WHERE QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) = @other_table
          AND c.name IN ('customer_id', 'full_name', 'email', 'phone');

        IF @has_cols >= 1
        BEGIN
            SET @sql = N'
                INSERT INTO #results (object_name, rows_found)
                SELECT 
                    N''' + @other_table + N''',
                    COUNT(*)
                FROM ' + @other_table + N'
                WHERE 1=0';
            
            -- Dynamically build WHERE clause based on available columns
            DECLARE @where_clause NVARCHAR(MAX) = N'';
            
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) = @other_table
                  AND c.name = 'customer_id'
            )
                SET @where_clause = @where_clause + N' OR customer_id = @customer_id';
                
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) = @other_table
                  AND c.name = 'full_name'
            )
                SET @where_clause = @where_clause + N' OR full_name = @full_name';
                
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) = @other_table
                  AND c.name = 'email'
            )
                SET @where_clause = @where_clause + N' OR email = @email';
                
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) = @other_table
                  AND c.name = 'phone'
            )
                SET @where_clause = @where_clause + N' OR phone = @phone';
            
            IF LEN(@where_clause) > 0
            BEGIN
                SET @where_clause = N' WHERE ' + STUFF(@where_clause, 1, 4, N'');
                SET @sql = N'
                    INSERT INTO #results (object_name, rows_found)
                    SELECT 
                        N''' + @other_table + N''',
                        COUNT(*)
                    FROM ' + @other_table + @where_clause;
                
                EXEC sp_executesql @sql, 
                    N'@customer_id INT, @full_name NVARCHAR(255), @email NVARCHAR(255), @phone NVARCHAR(255)',
                    @customer_id, @full_name, @email, @phone;
            END
            ELSE
            BEGIN
                INSERT INTO #results (object_name, rows_found)
                VALUES (@other_table, 0);
            END
        END

        FETCH NEXT FROM other_cursor INTO @other_table;
    END;

    CLOSE other_cursor;
    DEALLOCATE other_cursor;
END

-- Return final results
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
