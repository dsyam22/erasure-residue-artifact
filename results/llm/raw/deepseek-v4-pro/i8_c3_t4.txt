```sql
SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

-- Temporary table to hold results
CREATE TABLE #results (
    object_name nvarchar(255) NOT NULL,
    rows_found int NOT NULL
);

-- Check dbo.customer
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check for change data capture tables (CDC)
IF EXISTS (SELECT 1 FROM sys.databases WHERE name = DB_NAME() AND is_cdc_enabled = 1)
BEGIN
    DECLARE @cdc_table nvarchar(500);
    DECLARE cdc_cursor CURSOR FOR
        SELECT N'cdc.' + OBJECT_SCHEMA_NAME(ct.object_id) + '_' + OBJECT_NAME(ct.object_id) + '_CT'
        FROM cdc.change_tables ct
        WHERE OBJECT_NAME(ct.source_object_id) = 'customer'
          AND OBJECT_SCHEMA_NAME(ct.source_object_id) = 'dbo';
    
    OPEN cdc_cursor;
    FETCH NEXT FROM cdc_cursor INTO @cdc_table;
    
    WHILE @@FETCH_STATUS = 0
    BEGIN
        DECLARE @sql nvarchar(max) = N'INSERT INTO #results (object_name, rows_found) SELECT @obj, COUNT(*) FROM ' + QUOTENAME(@cdc_table) + 
            N' WHERE (customer_id = @cid OR full_name = @fn OR email = @em OR phone = @ph)';
        EXEC sp_executesql @sql, N'@obj nvarchar(255), @cid int, @fn nvarchar(255), @em nvarchar(255), @ph nvarchar(255)', 
            @obj = @cdc_table, @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone;
        FETCH NEXT FROM cdc_cursor INTO @cdc_table;
    END
    
    CLOSE cdc_cursor;
    DEALLOCATE cdc_cursor;
END

-- Check temporal history tables (if any exist for dbo.customer)
IF EXISTS (
    SELECT 1 
    FROM sys.tables t
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    WHERE t.name = 'customer' AND s.name = 'dbo' AND t.temporal_type = 2
)
BEGIN
    DECLARE @history_table nvarchar(255);
    SELECT @history_table = N'dbo.' + t.history_table_name
    FROM sys.tables t
    WHERE t.name = 'customer' AND t.schema_id = SCHEMA_ID('dbo') AND t.temporal_type = 2;
    
    IF @history_table IS NOT NULL
    BEGIN
        DECLARE @hist_sql nvarchar(max) = N'INSERT INTO #results (object_name, rows_found) SELECT @obj, COUNT(*) FROM ' + QUOTENAME(@history_table) + 
            N' WHERE (customer_id = @cid OR full_name = @fn OR email = @em OR phone = @ph)';
        EXEC sp_executesql @hist_sql, N'@obj nvarchar(255), @cid int, @fn nvarchar(255), @em nvarchar(255), @ph nvarchar(255)', 
            @obj = @history_table, @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone;
    END
END

-- Check any other user tables that might contain copies of this data
-- We'll look for tables that have columns named like the original columns
DECLARE @table_schema nvarchar(128);
DECLARE @table_name nvarchar(128);
DECLARE table_cursor CURSOR FOR
    SELECT DISTINCT s.name, t.name
    FROM sys.tables t
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    JOIN sys.columns c ON t.object_id = c.object_id
    WHERE c.name IN ('customer_id', 'full_name', 'email', 'phone')
      AND NOT (s.name = 'dbo' AND t.name = 'customer')
      AND t.temporal_type = 0  -- exclude temporal tables themselves, history handled above
      AND t.is_ms_shipped = 0;

OPEN table_cursor;
FETCH NEXT FROM table_cursor INTO @table_schema, @table_name;

WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @full_table nvarchar(255) = QUOTENAME(@table_schema) + N'.' + QUOTENAME(@table_name);
    DECLARE @check_sql nvarchar(max) = N'INSERT INTO #results (object_name, rows_found) SELECT @obj, COUNT(*) FROM ' + @full_table + 
        N' WHERE (customer_id = @cid OR full_name = @fn OR email = @em OR phone = @ph)';
    
    BEGIN TRY
        EXEC sp_executesql @check_sql, N'@obj nvarchar(255), @cid int, @fn nvarchar(255), @em nvarchar(255), @ph nvarchar(255)', 
            @obj = @table_schema + N'.' + @table_name, @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone;
    END TRY
    BEGIN CATCH
        -- If column doesn't exist in this table, skip silently
    END CATCH
    
    FETCH NEXT FROM table_cursor INTO @table_schema, @table_name;
END

CLOSE table_cursor;
DEALLOCATE table_cursor;

-- Return results
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
