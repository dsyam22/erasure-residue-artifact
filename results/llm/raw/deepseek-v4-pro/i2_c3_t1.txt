```sql
DECLARE @customer_id INT = 42;
DECLARE @full_name NVARCHAR(255) = N'Subject_0000042';
DECLARE @email NVARCHAR(255) = N'subject42@example.invalid';
DECLARE @phone NVARCHAR(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name NVARCHAR(255),
    rows_found INT
);

-- Check current table
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer',
    COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check temporal history table
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer_history',
    COUNT(*)
FROM dbo.customer_history
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC change tables (if any exist)
IF EXISTS (SELECT 1 FROM sys.tables WHERE name LIKE '%_CT' AND is_tracked_by_cdc = 1)
BEGIN
    DECLARE @ct_table NVARCHAR(255);
    DECLARE ct_cursor CURSOR FOR
        SELECT QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
        FROM sys.tables t
        JOIN sys.schemas s ON t.schema_id = s.schema_id
        WHERE t.name LIKE '%_CT'
          AND t.is_tracked_by_cdc = 1;

    OPEN ct_cursor;
    FETCH NEXT FROM ct_cursor INTO @ct_table;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        DECLARE @sql NVARCHAR(MAX) = 
            N'INSERT INTO #results (object_name, rows_found)
              SELECT @obj, COUNT(*)
              FROM ' + @ct_table + N'
              WHERE customer_id = @cid
                 OR full_name = @fn
                 OR email = @em
                 OR phone = @ph;';

        EXEC sp_executesql @sql,
            N'@obj NVARCHAR(255), @cid INT, @fn NVARCHAR(255), @em NVARCHAR(255), @ph NVARCHAR(255)',
            @obj = @ct_table, @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone;

        FETCH NEXT FROM ct_cursor INTO @ct_table;
    END;

    CLOSE ct_cursor;
    DEALLOCATE ct_cursor;
END

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
