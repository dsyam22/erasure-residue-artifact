```sql
SET NOCOUNT ON;

DECLARE @CustomerID INT = 42;
DECLARE @FullName NVARCHAR(255) = N'Subject_0000042';
DECLARE @Email NVARCHAR(255) = N'subject42@example.invalid';
DECLARE @Phone NVARCHAR(50) = N'+1-555-0000042';

-- Temporary table to store results
CREATE TABLE #Results (
    object_name NVARCHAR(255) NOT NULL,
    rows_found INT NOT NULL
);

-- Check dbo.customer
INSERT INTO #Results (object_name, rows_found)
SELECT 
    N'dbo.customer',
    COUNT(*)
FROM dbo.customer
WHERE customer_id = @CustomerID
   OR full_name = @FullName
   OR email = @Email
   OR phone = @Phone;

-- Check dbo.customer_history
INSERT INTO #Results (object_name, rows_found)
SELECT 
    N'dbo.customer_history',
    COUNT(*)
FROM dbo.customer_history
WHERE customer_id = @CustomerID
   OR full_name = @FullName
   OR email = @Email
   OR phone = @Phone;

-- Return results
SELECT object_name, rows_found
FROM #Results
ORDER BY object_name;

DROP TABLE #Results;
```
