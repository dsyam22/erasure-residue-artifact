```sql
SET NOCOUNT ON;

DECLARE @customer_id INT = 42;
DECLARE @full_name NVARCHAR(255) = N'Subject_0000042';
DECLARE @email NVARCHAR(255) = N'subject42@example.invalid';
DECLARE @phone NVARCHAR(255) = N'+1-555-0000042';

CREATE TABLE #Results (
    object_name NVARCHAR(255) NOT NULL,
    rows_found INT NOT NULL
);

-- Check the base table
INSERT INTO #Results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC change table
IF EXISTS (SELECT 1 FROM sys.tables t JOIN sys.schemas s ON t.schema_id = s.schema_id WHERE s.name = 'cdc' AND t.name = 'dbo_customer_CT')
BEGIN
    INSERT INTO #Results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', COUNT(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @customer_id
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END
ELSE
BEGIN
    INSERT INTO #Results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', 0;
END

-- Check temporal history tables (if any exist for dbo.customer)
IF EXISTS (SELECT 1 FROM sys.tables t JOIN sys.schemas s ON t.schema_id = s.schema_id WHERE s.name = 'dbo' AND t.name LIKE 'customer%history%')
BEGIN
    DECLARE @hist_table NVARCHAR(255);
    DECLARE hist_cursor CURSOR FOR
        SELECT QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
        FROM sys.tables t
        JOIN sys.schemas s ON t.schema_id = s.schema_id
        WHERE s.name = 'dbo'
          AND t.name LIKE 'customer%history%';

    OPEN hist_cursor;
    FETCH NEXT FROM hist_cursor INTO @hist_table;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        DECLARE @sql NVARCHAR(MAX) = N'INSERT INTO #Results (object_name, rows_found) SELECT @obj, COUNT(*) FROM ' + @hist_table + N' WHERE customer_id = @cid OR full_name = @fn OR email = @em OR phone = @ph';
        EXEC sp_executesql @sql, N'@obj NVARCHAR(255), @cid INT, @fn NVARCHAR(255), @em NVARCHAR(255), @ph NVARCHAR(255)', @obj = @hist_table, @cid = @customer_id, @fn = @full_name, @em = @email, @ph = @phone;
        FETCH NEXT FROM hist_cursor INTO @hist_table;
    END

    CLOSE hist_cursor;
    DEALLOCATE hist_cursor;
END
ELSE
BEGIN
    INSERT INTO #Results (object_name, rows_found)
    SELECT N'dbo.customer_history', 0;
END

SELECT object_name, rows_found
FROM #Results
ORDER BY object_name;

DROP TABLE #Results;
```
