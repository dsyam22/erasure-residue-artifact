```sql
SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(255) NOT NULL,
    rows_found int NOT NULL
);

-- Check current customer table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check current member table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.member', COUNT(*)
FROM dbo.member
WHERE member_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check member history table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.member_history', COUNT(*)
FROM dbo.member_history
WHERE member_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC capture instance
IF EXISTS (
    SELECT 1 
    FROM cdc.change_tables 
    WHERE capture_instance = N'dbo_customer'
)
BEGIN
    DECLARE @cdc_sql nvarchar(max) = N'
        SELECT COUNT(*)
        FROM cdc.dbo_customer_CT
        WHERE customer_id = @customer_id
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone';
    
    DECLARE @cdc_count int;
    
    EXEC sp_executesql @cdc_sql, 
        N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255)',
        @customer_id, @full_name, @email, @phone;
    
    -- sp_executesql with output parameter
    DECLARE @cdc_sql2 nvarchar(max) = N'
        SELECT @count = COUNT(*)
        FROM cdc.dbo_customer_CT
        WHERE customer_id = @customer_id
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone';
    
    DECLARE @count_out int;
    EXEC sp_executesql @cdc_sql2,
        N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255), @count int OUTPUT',
        @customer_id, @full_name, @email, @phone, @count_out OUTPUT;
    
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', @count_out;
END
ELSE
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', 0;
END

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
