```sql
SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(255) NOT NULL,
    rows_found int NOT NULL
);

-- Check current table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check temporal history table
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer_history', COUNT(*)
FROM dbo.customer_history
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check CDC change tables (if any exist)
DECLARE @cdc_table nvarchar(255);
DECLARE cdc_cursor CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name + N'.' + t.name
    FROM sys.tables t
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    WHERE t.is_tracked_by_cdc = 1;

OPEN cdc_cursor;
FETCH NEXT FROM cdc_cursor INTO @cdc_table;

WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @sql nvarchar(max) = N'
        INSERT INTO #results (object_name, rows_found)
        SELECT N''' + @cdc_table + N''', COUNT(*)
        FROM ' + QUOTENAME(PARSENAME(@cdc_table, 2)) + N'.' + QUOTENAME(PARSENAME(@cdc_table, 1)) + N'
        WHERE customer_id = @customer_id
           OR full_name = @full_name
           OR email = @email
           OR phone = @phone;';
    
    EXEC sp_executesql @sql,
         N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255)',
         @customer_id, @full_name, @email, @phone;
    
    FETCH NEXT FROM cdc_cursor INTO @cdc_table;
END;

CLOSE cdc_cursor;
DEALLOCATE cdc_cursor;

-- Check for any other user tables that might contain customer data (common patterns)
-- This covers downstream copies that may not be temporal or CDC-enabled
DECLARE @table_name nvarchar(255);
DECLARE table_cursor CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name + N'.' + t.name
    FROM sys.tables t
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    WHERE t.name LIKE '%customer%'
       OR t.name LIKE '%subject%'
       OR t.name LIKE '%personal%'
       OR t.name LIKE '%user%'
       OR t.name LIKE '%contact%'
       OR t.name LIKE '%profile%';

OPEN table_cursor;
FETCH NEXT FROM table_cursor INTO @table_name;

WHILE @@FETCH_STATUS = 0
BEGIN
    -- Skip tables we've already checked
    IF @table_name NOT IN (N'dbo.customer', N'dbo.customer_history')
    BEGIN
        -- Check if the table has relevant columns
        IF EXISTS (
            SELECT 1
            FROM sys.columns c
            JOIN sys.tables t ON c.object_id = t.object_id
            JOIN sys.schemas s ON t.schema_id = s.schema_id
            WHERE s.name + N'.' + t.name = @table_name
              AND c.name IN ('customer_id', 'full_name', 'email', 'phone')
        )
        BEGIN
            SET @sql = N'
                INSERT INTO #results (object_name, rows_found)
                SELECT N''' + @table_name + N''', COUNT(*)
                FROM ' + QUOTENAME(PARSENAME(@table_name, 2)) + N'.' + QUOTENAME(PARSENAME(@table_name, 1)) + N'
                WHERE 1=0';
            
            -- Build dynamic WHERE clause based on available columns
            DECLARE @where_clause nvarchar(max) = N'';
            
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE s.name + N'.' + t.name = @table_name AND c.name = 'customer_id'
            )
                SET @where_clause = @where_clause + N' OR customer_id = @customer_id';
            
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE s.name + N'.' + t.name = @table_name AND c.name = 'full_name'
            )
                SET @where_clause = @where_clause + N' OR full_name = @full_name';
            
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE s.name + N'.' + t.name = @table_name AND c.name = 'email'
            )
                SET @where_clause = @where_clause + N' OR email = @email';
            
            IF EXISTS (
                SELECT 1 FROM sys.columns c
                JOIN sys.tables t ON c.object_id = t.object_id
                JOIN sys.schemas s ON t.schema_id = s.schema_id
                WHERE s.name + N'.' + t.name = @table_name AND c.name = 'phone'
            )
                SET @where_clause = @where_clause + N' OR phone = @phone';
            
            IF LEN(@where_clause) > 0
            BEGIN
                SET @where_clause = RIGHT(@where_clause, LEN(@where_clause) - 4); -- Remove leading ' OR '
                SET @sql = N'
                    INSERT INTO #results (object_name, rows_found)
                    SELECT N''' + @table_name + N''', COUNT(*)
                    FROM ' + QUOTENAME(PARSENAME(@table_name, 2)) + N'.' + QUOTENAME(PARSENAME(@table_name, 1)) + N'
                    WHERE ' + @where_clause + N';';
                
                EXEC sp_executesql @sql,
                     N'@customer_id int, @full_name nvarchar(255), @email nvarchar(255), @phone nvarchar(255)',
                     @customer_id, @full_name, @email, @phone;
            END
            ELSE
            BEGIN
                -- No relevant columns found, insert 0
                INSERT INTO #results (object_name, rows_found)
                SELECT @table_name, 0;
            END
        END
    END
    
    FETCH NEXT FROM table_cursor INTO @table_name;
END;

CLOSE table_cursor;
DEALLOCATE table_cursor;

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
