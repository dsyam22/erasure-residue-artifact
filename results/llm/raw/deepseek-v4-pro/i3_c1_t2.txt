```sql
DECLARE @customer_id INT = 42;
DECLARE @full_name NVARCHAR(255) = N'Subject_0000042';
DECLARE @email NVARCHAR(255) = N'subject42@example.invalid';
DECLARE @phone NVARCHAR(255) = N'+1-555-0000042';

SELECT 
    o.object_name,
    COUNT(r.source_row) AS rows_found
FROM 
    (VALUES 
        (N'dbo.customer'),
        (N'dbo.member')
    ) AS o(object_name)
OUTER APPLY (
    SELECT 1 AS source_row
    FROM dbo.customer c
    WHERE o.object_name = N'dbo.customer'
      AND c.customer_id = @customer_id
      AND (c.full_name = @full_name OR c.email = @email OR c.phone = @phone)
    UNION ALL
    SELECT 1
    FROM dbo.member m
    WHERE o.object_name = N'dbo.member'
      AND m.member_id = @customer_id
      AND (m.full_name = @full_name OR m.email = @email OR m.phone = @phone)
) r
GROUP BY o.object_name
ORDER BY o.object_name;
```
