```sql
DECLARE @original_email NVARCHAR(255) = 'subject42@example.invalid';
DECLARE @original_phone NVARCHAR(50) = '+1-555-0000042';
DECLARE @original_name NVARCHAR(255) = 'Subject_0000042';

DECLARE @results TABLE (
    object_name NVARCHAR(255),
    rows_found INT
);

-- Check dbo.customer
INSERT INTO @results (object_name, rows_found)
SELECT 
    'dbo.customer',
    COUNT(*)
FROM dbo.customer
WHERE customer_id = 42
  AND (
      full_name = @original_name
      OR email = @original_email
      OR phone = @original_phone
  );

-- Check dbo.member
INSERT INTO @results (object_name, rows_found)
SELECT 
    'dbo.member',
    COUNT(*)
FROM dbo.member
WHERE member_id = 42
  AND (
      full_name = @original_name
      OR email = @original_email
      OR phone = @original_phone
  );

-- Check dbo.member_history
INSERT INTO @results (object_name, rows_found)
SELECT 
    'dbo.member_history',
    COUNT(*)
FROM dbo.member_history
WHERE member_id = 42
  AND (
      full_name = @original_name
      OR email = @original_email
      OR phone = @original_phone
  );

-- Check CDC change table for dbo.customer
INSERT INTO @results (object_name, rows_found)
SELECT 
    'cdc.dbo_customer_CT',
    COUNT(*)
FROM cdc.dbo_customer_CT
WHERE customer_id = 42
  AND (
      full_name = @original_name
      OR email = @original_email
      OR phone = @original_phone
  );

-- Return results
SELECT object_name, rows_found
FROM @results
ORDER BY object_name;
```
