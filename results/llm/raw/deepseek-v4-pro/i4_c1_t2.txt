```sql
SET NOCOUNT ON;

DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50) = N'+1-555-0000042';

-- Temp table to hold results
CREATE TABLE #results (
    object_name nvarchar(255),
    rows_found int
);

-- Check dbo.customer
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer',
    COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
  AND (full_name = @full_name OR email = @email OR phone = @phone);

-- Check dbo.member
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.member',
    COUNT(*)
FROM dbo.member
WHERE member_id = @customer_id
  AND (full_name = @full_name OR email = @email OR phone = @phone);

-- Also check for any rows in dbo.customer with the original data regardless of customer_id
-- (in case the ID was changed but the personal data remains)
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer (any id)',
    COUNT(*)
FROM dbo.customer
WHERE full_name = @full_name OR email = @email OR phone = @phone;

-- Same for dbo.member
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.member (any id)',
    COUNT(*)
FROM dbo.member
WHERE full_name = @full_name OR email = @email OR phone = @phone;

-- Return final result set
SELECT object_name, rows_found FROM #results;

DROP TABLE #results;
```
