```sql
DECLARE @customer_id INT = 42;
DECLARE @full_name NVARCHAR(255) = N'Subject_0000042';
DECLARE @email NVARCHAR(255) = N'subject42@example.invalid';
DECLARE @phone NVARCHAR(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name NVARCHAR(255),
    rows_found INT
);

-- Check dbo.customer
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'dbo.customer' AS object_name,
    COUNT(*) AS rows_found
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check reporting.customer_replica
INSERT INTO #results (object_name, rows_found)
SELECT 
    N'reporting.customer_replica' AS object_name,
    COUNT(*) AS rows_found
FROM reporting.customer_replica
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check for CDC change tables (if any exist)
IF EXISTS (SELECT 1 FROM sys.change_tracking_tables)
BEGIN
    DECLARE @ct_sql NVARCHAR(MAX) = N'';
    
    SELECT @ct_sql = @ct_sql + 
        N'INSERT INTO #results (object_name, rows_found) ' +
        N'SELECT N''' + s.name + N'.' + t.name + N''' AS object_name, ' +
        N'COUNT(*) AS rows_found ' +
        N'FROM ' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) + N' ct ' +
        N'WHERE ct.customer_id = @customer_id ' +
        N'   OR ct.full_name = @full_name ' +
        N'   OR ct.email = @email ' +
        N'   OR ct.phone = @phone; '
    FROM sys.change_tracking_tables ctt
    JOIN sys.tables t ON ctt.object_id = t.object_id
    JOIN sys.schemas s ON t.schema_id = s.schema_id;
    
    IF @ct_sql <> N''
    BEGIN
        EXEC sp_executesql @ct_sql, 
            N'@customer_id INT, @full_name NVARCHAR(255), @email NVARCHAR(255), @phone NVARCHAR(255)',
            @customer_id, @full_name, @email, @phone;
    END
END

-- Check for temporal history tables (if any exist)
IF EXISTS (SELECT 1 FROM sys.tables WHERE temporal_type_desc = 'HISTORY_TABLE')
BEGIN
    DECLARE @hist_sql NVARCHAR(MAX) = N'';
    
    SELECT @hist_sql = @hist_sql + 
        N'INSERT INTO #results (object_name, rows_found) ' +
        N'SELECT N''' + s.name + N'.' + t.name + N''' AS object_name, ' +
        N'COUNT(*) AS rows_found ' +
        N'FROM ' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) + N' h ' +
        N'WHERE h.customer_id = @customer_id ' +
        N'   OR h.full_name = @full_name ' +
        N'   OR h.email = @email ' +
        N'   OR h.phone = @phone; '
    FROM sys.tables t
    JOIN sys.schemas s ON t.schema_id = s.schema_id
    WHERE t.temporal_type_desc = 'HISTORY_TABLE';
    
    IF @hist_sql <> N''
    BEGIN
        EXEC sp_executesql @hist_sql, 
            N'@customer_id INT, @full_name NVARCHAR(255), @email NVARCHAR(255), @phone NVARCHAR(255)',
            @customer_id, @full_name, @email, @phone;
    END
END

-- Ensure we have at least one row for each expected object
IF NOT EXISTS (SELECT 1 FROM #results WHERE object_name = N'dbo.customer')
    INSERT INTO #results VALUES (N'dbo.customer', 0);

IF NOT EXISTS (SELECT 1 FROM #results WHERE object_name = N'reporting.customer_replica')
    INSERT INTO #results VALUES (N'reporting.customer_replica', 0);

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
