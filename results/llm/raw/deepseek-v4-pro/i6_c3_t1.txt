```sql
DECLARE @customer_id INT = 42;
DECLARE @phone NVARCHAR(50) = '+1-555-0000042';

CREATE TABLE #results (
    object_name NVARCHAR(255),
    rows_found INT
);

-- Check current dbo.customer table for the phone number
INSERT INTO #results (object_name, rows_found)
SELECT 'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
  AND EXISTS (
      SELECT 1
      FROM sys.columns c
      WHERE c.object_id = OBJECT_ID('dbo.customer')
        AND c.name = 'phone'
  )
  AND phone = @phone;

-- Check CDC change table for the phone number
INSERT INTO #results (object_name, rows_found)
SELECT 'cdc.dbo_customer_CT', COUNT(*)
FROM cdc.dbo_customer_CT
WHERE customer_id = @customer_id
  AND phone = @phone;

-- Check any other tables that might contain phone data (downstream copies)
-- Search for tables with a column named 'phone'
INSERT INTO #results (object_name, rows_found)
SELECT 
    s.name + '.' + t.name AS object_name,
    SUM(CASE WHEN c.name = 'phone' THEN 1 ELSE 0 END) AS rows_found
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
JOIN sys.columns c ON t.object_id = c.object_id
WHERE c.name = 'phone'
  AND NOT (s.name = 'dbo' AND t.name = 'customer')
  AND NOT (s.name = 'cdc' AND t.name = 'dbo_customer_CT')
GROUP BY s.name, t.name;

-- If no other tables with phone column, ensure at least the known objects are listed
IF NOT EXISTS (SELECT 1 FROM #results WHERE object_name = 'dbo.customer')
    INSERT INTO #results VALUES ('dbo.customer', 0);
IF NOT EXISTS (SELECT 1 FROM #results WHERE object_name = 'cdc.dbo_customer_CT')
    INSERT INTO #results VALUES ('cdc.dbo_customer_CT', 0);

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
