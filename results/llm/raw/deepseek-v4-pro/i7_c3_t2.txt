```sql
DECLARE @customer_id int = 42;
DECLARE @full_name nvarchar(255) = N'Subject_0000042';
DECLARE @email nvarchar(255) = N'subject42@example.invalid';
DECLARE @phone nvarchar(255) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(255) NOT NULL,
    rows_found int NOT NULL
);

-- Check dbo.customer
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer', COUNT(*)
FROM dbo.customer
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check reporting.customer_replica
INSERT INTO #results (object_name, rows_found)
SELECT N'reporting.customer_replica', COUNT(*)
FROM reporting.customer_replica
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Check for Change Data Capture (CDC) tables
IF EXISTS (SELECT 1 FROM sys.tables t
           INNER JOIN sys.schemas s ON t.schema_id = s.schema_id
           WHERE t.name LIKE 'customer%' AND t.is_tracked_by_cdc = 1)
BEGIN
    -- CDC enabled tables have a corresponding change table in the 'cdc' schema
    INSERT INTO #results (object_name, rows_found)
    SELECT N'cdc.dbo_customer_CT', COUNT(*)
    FROM cdc.dbo_customer_CT
    WHERE customer_id = @customer_id
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END

-- Check for temporal history tables (system-versioned)
IF EXISTS (SELECT 1 FROM sys.tables t
           INNER JOIN sys.schemas s ON t.schema_id = s.schema_id
           WHERE t.name = 'customer' AND t.temporal_type = 2)
BEGIN
    INSERT INTO #results (object_name, rows_found)
    SELECT N'dbo.customer_history', COUNT(*)
    FROM dbo.customer_history
    WHERE customer_id = @customer_id
       OR full_name = @full_name
       OR email = @email
       OR phone = @phone;
END

-- Include any other possible tables that might contain the data
-- (e.g., audit tables, log tables, backup tables)
INSERT INTO #results (object_name, rows_found)
SELECT N'dbo.customer_audit', COUNT(*)
FROM dbo.customer_audit
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

INSERT INTO #results (object_name, rows_found)
SELECT N'reporting.customer_replica_audit', COUNT(*)
FROM reporting.customer_replica_audit
WHERE customer_id = @customer_id
   OR full_name = @full_name
   OR email = @email
   OR phone = @phone;

-- Return final result set
SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
