```sql
SELECT
    'dbo.customer' AS object_name,
    COUNT(1) AS rows_found
UNION ALL
SELECT
    'dbo.customer_history' AS object_name,
    COUNT(1)
WHERE EXISTS (
    SELECT 1
    FROM dbo.customer_history ch
    WHERE ch.customer_id = 42
      AND ch.full_name = 'Subject_0000042'
      AND ch.email = 'subject42@example.invalid'
      AND ch.phone = '+1-555-0000042'
)
```
