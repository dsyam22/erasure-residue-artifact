```sql
SELECT 'dbo.customer' AS object_name, COUNT(T1.*) AS rows_found
FROM dbo.customer T1
WHERE 
    T1.full_name = N'Subject_0000042' OR
    T1.email = N'subject42@example.invalid' OR
    T1.phone = N'+1-555-0000042'

UNION ALL

SELECT 'dbo.member', COUNT(T2.*) 
FROM dbo.member T2
WHERE 
    T2.full_name = N'Subject_0000042' OR
    T2.email = N'subject42@example.invalid' OR
    T2.phone = N'+1-555-0000042';
```
