```sql
DECLARE @RowCount INT;

-- Check dbo.customer for any remaining records matching the subject's details
SET @RowCount = (
    SELECT COUNT(1)
    FROM dbo.customer
    WHERE customer_id = 42
      AND full_name = 'Subject_0000042'
      AND email = 'subject42@example.invalid'
      AND phone = '+1-555-0000042'
);

SELECT
    'dbo.customer' AS object_name,
    CAST(@RowCount AS INT) AS rows_found;
```
