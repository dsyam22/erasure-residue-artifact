```sql
SELECT
    'dbo.customer' AS object_name,
    CAST((
        SELECT COUNT(*)
        FROM dbo.customer c
        WHERE 
            c.full_name = 'Subject_0000042' 
            OR c.email = 'subject42@example.invalid' 
            OR c.phone = '+1-555-0000042'
            OR c.customer_id = 42
    ) AS INT) AS rows_found;
```
