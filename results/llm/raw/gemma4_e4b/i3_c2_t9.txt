DECLARE @TargetName NVARCHAR(256) = 'Subject_0000042';
DECLARE @TargetEmail NVARCHAR(256) = 'subject42@example.invalid';
DECLARE @TargetPhone NVARCHAR(256) = '+1-555-0000042';

-- Check 1: dbo.customer (Base Table)
SELECT
    'dbo.customer' AS object_name,
    COUNT(CASE WHEN full_name = @TargetName OR email = @TargetEmail OR phone = @TargetPhone THEN 1 END) AS rows_found
FROM dbo.customer

UNION ALL

-- Check 2: cdc.dbo_customer_CT (CDC Change Table)
SELECT
    'cdc.dbo_customer_CT' AS object_name,
    COUNT(CASE WHEN full_name = @TargetName OR email = @TargetEmail OR phone = @TargetPhone THEN 1 END)
FROM cdc.dbo_customer_CT

UNION ALL

-- Check 3: dbo.member (Base Table - Current Version)
SELECT
    'dbo.member' AS object_name,
    COUNT(CASE WHEN full_name = @TargetName OR email = @TargetEmail OR phone = @TargetPhone THEN 1 END)
FROM dbo.member

UNION ALL
