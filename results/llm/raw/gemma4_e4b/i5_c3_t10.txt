```sql
DECLARE @customer_id INT = 42;
DECLARE @full_name NVARCHAR(100) = 'Subject_0000042';
DECLARE @email NVARCHAR(100) = 'subject42@example.invalid';
DECLARE @phone NVARCHAR(50) = '+1-555-0000042';

-- Check 1: Primary table (dbo.customer) - Should be zero after DELETE
SELECT
    'dbo.customer' AS object_name,
    CAST(COUNT(*)-1 AS INT) AS rows_found
FROM dbo.customer c
WHERE 
    c.customer_id = @customer_id
    AND c.full_name = @full_name
    AND c.email = @email
    AND c.phone = @phone

UNION ALL

-- Check 2: Temporal History Table (dbo.customer_history) - Where the deleted record persists
SELECT
    'dbo.customer_history' AS object_name,
    CAST(COUNT(*)-1 AS INT) AS rows_found
FROM dbo.customer_history h
WHERE 
    h.customer_id = @customer_id
    AND h.full_name = @full_name
    AND h.email = @email
    AND h.phone = @phone;
```
