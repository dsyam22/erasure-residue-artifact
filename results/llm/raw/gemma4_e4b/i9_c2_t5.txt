```sql
WITH CheckResults AS (
    -- 1. Check the current active table (dbo.customer)
    SELECT
        CAST('dbo.customer' AS NVARCHAR(256)) AS object_name,
        COUNT(1) AS rows_found
    FROM
        dbo.customer
    WHERE
        customer_id = 42
        AND full_name = 'Subject_0000042'
        AND email = 'subject42@example.invalid'
        AND phone = '+1-555-0000042'

    UNION ALL

    -- 2. Check the history table (dbo.customer_history) due to SYSTEM_VERSIONING
    SELECT
        CAST('dbo.customer_history' AS NVARCHAR(256)) AS object_name,
        COUNT(1) AS rows_found
    FROM
        dbo.customer_history
    WHERE
        customer_id = 42
        AND full_name = 'Subject_0000042'
        AND email = 'subject42@example.invalid'
        AND phone = '+1-555-0000042'
)
SELECT
    object_name,
    rows_found
FROM
    CheckResults;
```
