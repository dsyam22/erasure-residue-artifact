```sql
WITH CustomerCheck AS (
    -- Check dbo.customer table for residual data
    SELECT
        N'dbo.customer' AS object_name,
        CAST(COUNT(1) AS BIGINT) AS rows_found
    FROM
        dbo.customer
    WHERE
        customer_id = 42
        OR full_name = 'Subject_0000042'
        OR email = 'subject42@example.invalid'
        OR phone = '+1-555-0000042'
),
HistoryCheck AS (
    -- Check dbo.customer_history table for residual data
    SELECT
        N'dbo.customer_history' AS object_name,
        CAST(COUNT(1) AS BIGINT) AS rows_found
    FROM
        dbo.customer_history
    WHERE
        customer_id = 42
        OR full_name = 'Subject_0000042'
        OR email = 'subject42@example.invalid'
        OR phone = '+1-555-0000042'
)
-- Combine the results into a single result set
SELECT object_name, rows_found FROM CustomerCheck
UNION ALL
SELECT object_name, rows_found FROM HistoryCheck;
```
