```sql
SELECT 'dbo.customer' AS object_name, 
       ISNULL((
           SELECT COUNT(*) 
           FROM dbo.customer 
           WHERE full_name = 'Subject_0000042' OR email = 'subject42@example.invalid' OR phone = '+1-555-0000042'
       ), 0) AS rows_found

UNION ALL

SELECT 'dbo.member', 
       ISNULL((
           SELECT COUNT(*) 
           FROM dbo.member 
           WHERE full_name = 'Subject_0000042' OR email = 'subject42@example.invalid' OR phone = '+1-555-0000042'
       ), 0) AS rows_found;
```
