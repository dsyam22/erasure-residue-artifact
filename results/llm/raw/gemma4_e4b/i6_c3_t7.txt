```sql
SELECT object_name, rows_found FROM (
    -- Check 1: Primary Table (dbo.customer)
    SELECT 
        'dbo.customer' AS object_name,
        CAST(COUNT(CASE WHEN phone = '+1-555-0000042' THEN 1 END) AS INT) AS rows_found
    FROM dbo.customer
    WHERE customer_id = 42

    UNION ALL

    -- Check 2: CDC Change Table (cdc.dbo_customer_CT)
    SELECT 
        'cdc.dbo_customer_CT' AS object_name,
        CAST(COUNT(CASE WHEN phone = '+1-555-0000042' THEN 1 END) AS INT) AS rows_found
    FROM cdc.dbo_customer_CT
    WHERE customer_id = 42

) AS Results;
```
