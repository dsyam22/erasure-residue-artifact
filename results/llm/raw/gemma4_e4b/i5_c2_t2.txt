```sql
DECLARE @SubjectID INT = 42;
DECLARE @FullName NVARCHAR(100) = 'Subject_0000042';

SELECT
    'dbo.customer' AS object_name,
    COUNT(T.*) AS rows_found
FROM
    dbo.customer T
WHERE
    T.customer_id = @SubjectID
    AND T.full_name = @FullName
GROUP BY
    1 -- Dummy grouping to ensure COUNT works correctly when combined with UNION ALL

UNION ALL

SELECT
    'dbo.customer_history' AS object_name,
    COUNT(T.*) AS rows_found
FROM
    dbo.customer_history T
WHERE
    T.customer_id = @SubjectID
    AND T.full_name = @FullName;
```
