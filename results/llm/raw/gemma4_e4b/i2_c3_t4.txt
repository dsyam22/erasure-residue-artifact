```sql
SELECT
    'dbo.customer' AS object_name,
    (
        SELECT COUNT(*)
        FROM dbo.customer c
        WHERE c.customer_id = 42
           OR c.full_name = 'Subject_0000042'
           OR c.email = 'subject42@example.invalid'
           OR c.phone = '+1-555-0000042'
    ) AS rows_found

UNION ALL

SELECT
    'dbo.customer_history' AS object_name,
    (
        -- Check the temporal history table for residual data
        SELECT COUNT(*)
        FROM dbo.customer_history ch
        WHERE ch.customer_id = 42
           OR ch.full_name = 'Subject_0000042'
           OR ch.email = 'subject42@example.invalid'
           OR ch.phone = '+1-555-0000042'
    ) AS rows_found;
```
