WITH CustomerCheck AS (
    SELECT 
        'dbo.customer' AS object_name, 
        CAST(COUNT(*) AS INT) AS rows_found
    FROM dbo.customer
    WHERE full_name = 'Subject_0000042'
      AND email = 'subject42@example.invalid'
      AND phone = '+1-555-0000042'
),
CustomerCdcCheck AS (
    SELECT 
        'cdc.dbo_customer_CT' AS object_name, 
        CAST(COUNT(*) AS INT) AS rows_found
    FROM cdc.dbo_customer_CT
    WHERE full_name = 'Subject_0000042'
      AND email = 'subject42@example.invalid'
      AND phone = '+1-555-0000042'
),
MemberCheck AS (
    SELECT 
        'dbo.member' AS object_name, 
        CAST(COUNT(*) AS INT) AS rows_found
    FROM dbo.member
    WHERE full_name = 'Subject_0000042'
      AND email = 'subject42@example.invalid'
      AND phone = '+1-555-0000042'
),
MemberHistoryCheck AS (
    SELECT 
        'dbo.member_history' AS object_name, 
        CAST(COUNT(*) AS INT) AS rows_found
    FROM dbo.member_history
    WHERE full_name = 'Subject_0000042'
      AND email = 'subject42@example.invalid'
      AND phone = '+1-555-0000042'
)
SELECT 
    object_name, 
    rows_found
FROM CustomerCheck

UNION ALL

SELECT 
    object_name, 
    rows_found
FROM CustomerCdcCheck

UNION ALL

SELECT 
    object_name, 
    rows_found
FROM MemberCheck

UNION ALL

SELECT 
    object_name, 
    rows_found
FROM MemberHistoryCheck;
