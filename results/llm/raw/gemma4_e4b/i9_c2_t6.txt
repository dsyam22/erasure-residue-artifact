```sql
SELECT object_name, rows_found FROM (
    -- Check 1: Primary Table (dbo.customer) - Should be empty after DELETE
    SELECT
        'dbo.customer' AS object_name,
        CAST(COUNT(*) AS INT) AS rows_found
    FROM
        dbo.customer
    WHERE
        customer_id = 42
        AND full_name = 'Subject_0000042'
        AND email = 'subject42@example.invalid'
        AND phone = '+1-555-0000042'

    UNION ALL

    -- Check 2: History Table (dbo.customer_history) - Where temporal deletion records are stored
    SELECT
        'dbo.customer_history' AS object_name,
        CAST(COUNT(*) AS INT) AS rows_found
    FROM
        dbo.customer_history
    WHERE
        customer_id = 42
        AND full_name = 'Subject_0000042'
        AND email = 'subject42@example.invalid'
        AND phone = '+1-555-0000042'
) AS Result;
```
