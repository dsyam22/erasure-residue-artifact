```sql
-- SCRIPT
SELECT N'dbo.customer' AS object_name, COUNT_BIG(*) AS rows_found
FROM dbo.customer
WHERE CHARINDEX(N'+1-555-0000042', full_name COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', email COLLATE Latin1_General_100_BIN2) > 0

UNION ALL

SELECT N'cdc.dbo_customer_CT', COUNT_BIG(*)
FROM cdc.dbo_customer_CT
WHERE CHARINDEX(N'+1-555-0000042', phone COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', full_name COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', email COLLATE Latin1_General_100_BIN2) > 0

UNION ALL

SELECT N'cdc.ddl_history', COUNT_BIG(*)
FROM cdc.ddl_history
WHERE CHARINDEX(N'+1-555-0000042', ddl_command COLLATE Latin1_General_100_BIN2) > 0

UNION ALL

SELECT N'cdc.captured_columns', COUNT_BIG(*)
FROM cdc.captured_columns
WHERE CHARINDEX(N'+1-555-0000042', column_name COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', column_type COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', masking_function COLLATE Latin1_General_100_BIN2) > 0

UNION ALL

SELECT N'cdc.change_tables', COUNT_BIG(*)
FROM cdc.change_tables
WHERE CHARINDEX(N'+1-555-0000042', capture_instance COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', role_name COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', index_name COLLATE Latin1_General_100_BIN2) > 0
   OR CHARINDEX(N'+1-555-0000042', filegroup_name COLLATE Latin1_General_100_BIN2) > 0

UNION ALL

SELECT N'cdc.index_columns', COUNT_BIG(*)
FROM cdc.index_columns
WHERE CHARINDEX(N'+1-555-0000042', column_name COLLATE Latin1_General_100_BIN2) > 0
ORDER BY object_name;
```
