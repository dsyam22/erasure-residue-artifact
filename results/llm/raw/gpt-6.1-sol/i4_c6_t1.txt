SELECT N'dbo.customer' AS object_name, COUNT_BIG(*) AS rows_found
FROM dbo.customer
WHERE full_name = N'Subject_0000042'
   OR email = N'subject42@example.invalid'
   OR phone = N'+1-555-0000042'

UNION ALL

SELECT N'dbo.member', COUNT_BIG(*)
FROM dbo.member
WHERE full_name = N'Subject_0000042'
   OR email = N'subject42@example.invalid'
   OR phone = N'+1-555-0000042'

UNION ALL

-- Read the history table directly to include every stored version,
-- including versions with zero-duration validity intervals.
SELECT N'dbo.member_history', COUNT_BIG(*)
FROM dbo.member_history
WHERE full_name = N'Subject_0000042'
   OR email = N'subject42@example.invalid'
   OR phone = N'+1-555-0000042'

UNION ALL

-- Include all retained CDC operations, particularly update before-images.
SELECT N'cdc.dbo_customer_CT', COUNT_BIG(*)
FROM cdc.dbo_customer_CT
WHERE full_name = N'Subject_0000042'
   OR email = N'subject42@example.invalid'
   OR phone = N'+1-555-0000042'
ORDER BY object_name;
