-- Checks currently queryable rows in every catalog-visible table, including CDC.
-- Zero counts do not establish physical erasure from transaction logs, deallocated
-- pages, backups, or other storage not exposed as table rows.
-- CDC capture is asynchronous; changes not yet captured can affect the result.

DECLARE @sql nvarchar(max);

SELECT @sql =
    STRING_AGG(
        CONVERT(nvarchar(max),
            N'SELECT N''' +
            REPLACE(s.name + N'.' + t.name, N'''', N'''''') +
            N''' AS object_name, COUNT_BIG(*) AS rows_found FROM ' +
            QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) +
            N' AS src WHERE ' + COALESCE(p.predicate, N'1 = 0')
        ),
        N' UNION ALL '
    )
FROM sys.tables AS t
JOIN sys.schemas AS s
    ON s.schema_id = t.schema_id
OUTER APPLY
(
    SELECT STRING_AGG(CONVERT(nvarchar(max),
        CASE
            WHEN c.name COLLATE Latin1_General_100_CI_AS = N'customer_id'
            THEN
                N'TRY_CONVERT(nvarchar(max), src.' + QUOTENAME(c.name) +
                N') COLLATE Latin1_General_100_BIN2 = N''42'''
            ELSE
                N'(' +
                N'CHARINDEX(N''Subject_0000042'', TRY_CONVERT(nvarchar(max), src.' +
                QUOTENAME(c.name) + N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                N'CHARINDEX(N''subject42@example.invalid'', TRY_CONVERT(nvarchar(max), src.' +
                QUOTENAME(c.name) + N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                N'CHARINDEX(N''+1-555-0000042'', TRY_CONVERT(nvarchar(max), src.' +
                QUOTENAME(c.name) + N') COLLATE Latin1_General_100_BIN2) > 0)'
        END
    ), N' OR ') AS predicate
    FROM sys.columns AS c
    WHERE c.object_id = t.object_id
      AND c.encryption_type IS NULL
      AND
      (
          -- Text, XML, and SQL_VARIANT columns can hold copies of the originals.
          c.system_type_id IN (35, 99, 167, 175, 231, 239, 241, 98)
          OR
          (
              c.name COLLATE Latin1_General_100_CI_AS = N'customer_id'
              AND c.system_type_id IN
                  (48, 52, 56, 127, 106, 108, 59, 62, 60, 122)
          )
      )
) AS p;

IF @sql IS NULL
    SELECT
        CAST(NULL AS nvarchar(517)) AS object_name,
        CAST(NULL AS bigint) AS rows_found
    WHERE 1 = 0;
ELSE
BEGIN
    SET @sql =
        N'SELECT object_name, rows_found FROM (' + @sql +
        N') AS results ORDER BY object_name;';
    EXEC sys.sp_executesql @sql;
END;
