```sql
USE [llm_i2];

DECLARE @sql nvarchar(max);

-- Scan current tables, temporal history tables, and other user tables,
-- including CDC tables when present. Indexed views are checked as well.
SELECT @sql =
    STRING_AGG(
        CAST(
            N'SELECT N''' +
            REPLACE(o.schema_name + N'.' + o.object_name, N'''', N'''''') +
            N''' AS object_name, COUNT_BIG(*) AS rows_found FROM ' +
            QUOTENAME(o.schema_name) + N'.' + QUOTENAME(o.object_name) +
            N' WHERE ' + COALESCE(p.predicate_sql, N'1 = 0')
            AS nvarchar(max)
        ),
        N' UNION ALL '
    )
FROM
(
    SELECT t.object_id, s.name AS schema_name, t.name AS object_name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
       OR s.name = N'cdc'

    UNION ALL

    SELECT v.object_id, s.name, v.name
    FROM sys.views AS v
    JOIN sys.schemas AS s ON s.schema_id = v.schema_id
    WHERE EXISTS
    (
        SELECT 1
        FROM sys.indexes AS i
        WHERE i.object_id = v.object_id
          AND i.index_id = 1
    )
) AS o
OUTER APPLY
(
    SELECT STRING_AGG(
        CAST(
            CASE
                WHEN c.system_type_id IN
                     (35, 99, 167, 175, 231, 239, 241, 98)
                THEN
                    N'(CHARINDEX(N''Subject_0000042'', ' +
                    N'TRY_CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                    N'CHARINDEX(N''subject42@example.invalid'', ' +
                    N'TRY_CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                    N'CHARINDEX(N''+1-555-0000042'', ' +
                    N'TRY_CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0)'
                ELSE N'1 = 0'
            END +
            CASE
                WHEN c.name COLLATE Latin1_General_100_CI_AS = N'customer_id'
                  OR c.name COLLATE Latin1_General_100_CI_AS LIKE N'%[_]customer_id'
                THEN
                    N' OR TRY_CONVERT(nvarchar(max), ' +
                    QUOTENAME(c.name) + N') = N''42'''
                ELSE N''
            END
            AS nvarchar(max)
        ),
        N' OR '
    ) AS predicate_sql
    FROM sys.columns AS c
    WHERE c.object_id = o.object_id
      AND
      (
          c.system_type_id IN (35, 99, 167, 175, 231, 239, 241, 98)
          OR c.name COLLATE Latin1_General_100_CI_AS = N'customer_id'
          OR c.name COLLATE Latin1_General_100_CI_AS LIKE N'%[_]customer_id'
      )
) AS p;

SELECT @sql =
    CASE
        WHEN @sql IS NULL THEN
            N'SELECT CAST(NULL AS nvarchar(517)) AS object_name,
                     CAST(NULL AS bigint) AS rows_found
              WHERE 1 = 0;'
        ELSE
            N'SELECT object_name, rows_found FROM (' + @sql +
            N') AS results ORDER BY object_name;'
    END;

EXEC sys.sp_executesql @sql;
```
