```sql
USE [llm_i6];

DECLARE @phone nvarchar(100) = N'+1-555-0000042';
DECLARE @sql nvarchar(max);

SELECT @sql =
    STRING_AGG(
        CAST(
            N'SELECT N''' +
            REPLACE(QUOTENAME(s.name) + N'.' + QUOTENAME(t.name), N'''', N'''''') +
            N''' AS object_name, COUNT_BIG(*) AS rows_found FROM ' +
            QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) +
            N' WHERE ' + COALESCE(p.predicate, N'1 = 0')
            AS nvarchar(max)
        ),
        N' UNION ALL '
    )
FROM sys.tables AS t
JOIN sys.schemas AS s
    ON s.schema_id = t.schema_id
OUTER APPLY
(
    SELECT STRING_AGG(
        CAST(
            CASE
                WHEN c.system_type_id IN (34, 165, 173, 189)
                THEN
                    N'(CHARINDEX(@phone COLLATE Latin1_General_100_BIN2, ' +
                    N'TRY_CONVERT(nvarchar(max), CONVERT(varbinary(max), ' +
                    QUOTENAME(c.name) +
                    N')) COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                    N'CHARINDEX(@phone COLLATE Latin1_General_100_BIN2, ' +
                    N'CONVERT(nvarchar(max), CONVERT(varchar(max), ' +
                    N'CONVERT(varbinary(max), ' + QUOTENAME(c.name) +
                    N'))) COLLATE Latin1_General_100_BIN2) > 0)'
                ELSE
                    N'CHARINDEX(@phone COLLATE Latin1_General_100_BIN2, ' +
                    N'TRY_CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0'
            END AS nvarchar(max)
        ),
        N' OR '
    ) AS predicate
    FROM sys.columns AS c
    WHERE c.object_id = t.object_id
      AND c.system_type_id IN
          (34, 35, 98, 99, 165, 167, 173, 175, 189, 231, 239, 241)
) AS p
WHERE t.is_ms_shipped = 0;

SELECT @sql =
    CASE
        WHEN @sql IS NULL THEN
            N'SELECT CAST(NULL AS nvarchar(517)) AS object_name, ' +
            N'CAST(NULL AS bigint) AS rows_found WHERE 1 = 0;'
        ELSE
            N'SELECT object_name, rows_found FROM (' + @sql +
            N') AS checked_objects ORDER BY object_name;'
    END;

EXEC sys.sp_executesql
    @sql,
    N'@phone nvarchar(100)',
    @phone = @phone;
```
