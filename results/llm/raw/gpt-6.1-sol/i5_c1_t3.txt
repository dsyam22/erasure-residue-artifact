```sql
DECLARE @sql nvarchar(max);

-- Enumerate all user tables, including temporal history tables.
-- Search textual/XML columns for the original personal-data values,
-- and customer_id columns for the original identifier.
SELECT @sql =
    STRING_AGG(
        CONVERT(nvarchar(max),
            N'SELECT N''' +
            REPLACE(s.name + N'.' + t.name, N'''', N'''''') +
            N''' AS object_name, COUNT_BIG(*) AS rows_found FROM ' +
            N'[llm_i5].' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) +
            N' WHERE ' + COALESCE(p.predicate_sql, N'1 = 0')
        ),
        N' UNION ALL '
    )
FROM [llm_i5].sys.tables AS t
JOIN [llm_i5].sys.schemas AS s
    ON s.schema_id = t.schema_id
OUTER APPLY
(
    SELECT STRING_AGG(
        CONVERT(nvarchar(max),
            CASE
                WHEN c.system_type_id IN
                     (35, 99, 167, 175, 231, 239, 241)
                THEN
                    N'(CHARINDEX(N''Subject_0000042'', ' +
                    N'CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                    N'CHARINDEX(N''subject42@example.invalid'', ' +
                    N'CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
                    N'CHARINDEX(N''+1-555-0000042'', ' +
                    N'CONVERT(nvarchar(max), ' + QUOTENAME(c.name) +
                    N') COLLATE Latin1_General_100_BIN2) > 0' +
                    CASE WHEN c.name = N'customer_id'
                         THEN N' OR TRY_CONVERT(bigint, CONVERT(nvarchar(max), ' +
                              QUOTENAME(c.name) + N')) = 42'
                         ELSE N''
                    END + N')'
                WHEN c.name = N'customer_id'
                     AND c.system_type_id IN
                         (48, 52, 56, 59, 60, 62, 106, 108, 122, 127)
                THEN QUOTENAME(c.name) + N' = 42'
            END
        ),
        N' OR '
    ) AS predicate_sql
    FROM [llm_i5].sys.columns AS c
    WHERE c.object_id = t.object_id
) AS p
WHERE t.is_ms_shipped = 0;

SELECT @sql =
    N'SELECT object_name, rows_found FROM (' +
    COALESCE(
        @sql,
        N'SELECT CAST(NULL AS nvarchar(257)) AS object_name, ' +
        N'CAST(0 AS bigint) AS rows_found WHERE 1 = 0'
    ) +
    N') AS findings ORDER BY object_name;';

EXEC sys.sp_executesql @sql;
```
