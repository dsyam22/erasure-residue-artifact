```sql
USE [llm_i5];

SELECT
    N'dbo.customer' AS object_name,
    COUNT_BIG(*) AS rows_found
FROM dbo.customer
WHERE customer_id = 42
   OR full_name LIKE N'%Subject[_]0000042%'
   OR email LIKE N'%subject42@example.invalid%'
   OR phone LIKE N'%+1-555-0000042%'

UNION ALL

SELECT
    N'dbo.customer_history' AS object_name,
    COUNT_BIG(*) AS rows_found
FROM dbo.customer_history
WHERE customer_id = 42
   OR full_name LIKE N'%Subject[_]0000042%'
   OR email LIKE N'%subject42@example.invalid%'
   OR phone LIKE N'%+1-555-0000042%';
```
