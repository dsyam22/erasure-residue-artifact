DECLARE @sql nvarchar(max);

SELECT @sql =
    STRING_AGG(
        CAST(
            N'SELECT N''' +
            REPLACE(s.name + N'.' + t.name, N'''', N'''''') +
            N''' AS object_name, COUNT_BIG(*) AS rows_found FROM ' +
            N'[llm_i4].' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name) +
            N' AS src WHERE ' + COALESCE(p.predicate_sql, N'1 = 0')
            AS nvarchar(max)
        ),
        N' UNION ALL '
    )
FROM [llm_i4].sys.tables AS t
JOIN [llm_i4].sys.schemas AS s
    ON s.schema_id = t.schema_id
OUTER APPLY
(
    SELECT STRING_AGG(
        CAST(
            N'(' +
            N'CHARINDEX(@original_name, TRY_CONVERT(nvarchar(max), src.' +
            QUOTENAME(c.name) + N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
            N'CHARINDEX(@original_email, TRY_CONVERT(nvarchar(max), src.' +
            QUOTENAME(c.name) + N') COLLATE Latin1_General_100_BIN2) > 0 OR ' +
            N'CHARINDEX(@original_phone, TRY_CONVERT(nvarchar(max), src.' +
            QUOTENAME(c.name) + N') COLLATE Latin1_General_100_BIN2) > 0)'
            AS nvarchar(max)
        ),
        N' OR '
    ) AS predicate_sql
    FROM [llm_i4].sys.columns AS c
    WHERE c.object_id = t.object_id
      AND c.system_type_id IN
          (35, 99, 167, 175, 231, 239, 241, 98)
          -- text, ntext, varchar, char, nvarchar, nchar, xml, sql_variant
) AS p;

-- Scan physical history and CDC tables directly, including retained versions.
-- Each row is counted once even if several columns contain original values.
-- This checks SQL-visible table contents, not transaction-log remnants,
-- deallocated pages, backups, or external copies.
SELECT @sql =
    N'SELECT object_name, rows_found FROM (' + @sql +
    N') AS findings ORDER BY object_name;';

EXEC sys.sp_executesql
    @sql,
    N'@original_name nvarchar(100),
      @original_email nvarchar(320),
      @original_phone nvarchar(100)',
    @original_name = N'Subject_0000042',
    @original_email = N'subject42@example.invalid',
    @original_phone = N'+1-555-0000042';
