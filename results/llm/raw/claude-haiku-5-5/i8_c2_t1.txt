```sql
SET NOCOUNT ON;

-- Original personal data values of the erased subject (customer_id 42)
CREATE TABLE #needles (val nvarchar(200) NOT NULL);
INSERT INTO #needles (val) VALUES
    (N'Subject_0000042'),
    (N'subject42@example.invalid'),
    (N'+1-555-0000042');

CREATE TABLE #results (
    object_name nvarchar(300) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname, @obj nvarchar(300),
        @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl CURSOR LOCAL FAST_FORWARD FOR
    SELECT TABLE_SCHEMA, TABLE_NAME
    FROM INFORMATION_SCHEMA.TABLES
    WHERE TABLE_TYPE = N'BASE TABLE'
    ORDER BY TABLE_SCHEMA, TABLE_NAME;

OPEN tbl;
FETCH NEXT FROM tbl INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj  = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @pred = NULL;
    SET @cnt  = 0;

    -- Build a predicate: any character column contains any needle value
    SELECT @pred = STRING_AGG(
               CAST(N'CHARINDEX(N''' + REPLACE(n.val, N'''', N'''''') + N''', '
                    + QUOTENAME(c.COLUMN_NAME)
                    + N' COLLATE DATABASE_DEFAULT) > 0' AS nvarchar(max)),
               N' OR ')
    FROM INFORMATION_SCHEMA.COLUMNS AS c
    CROSS JOIN #needles AS n
    WHERE c.TABLE_SCHEMA = @schema
      AND c.TABLE_NAME   = @table
      AND c.DATA_TYPE IN (N'char', N'varchar', N'nchar', N'nvarchar');

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @c = COUNT(*) FROM ' + @obj + N' WHERE ' + @pred + N';';
        EXEC sys.sp_executesql @sql, N'@c bigint OUTPUT', @c = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (@schema + N'.' + @table, ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl INTO @schema, @table;
END

CLOSE tbl;
DEALLOCATE tbl;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
DROP TABLE #needles;
```
