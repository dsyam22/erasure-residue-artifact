```sql
SET NOCOUNT ON;

DECLARE @full  nvarchar(200) = N'Subject_0000042';
DECLARE @email nvarchar(200) = N'subject42@example.invalid';
DECLARE @phone nvarchar(50)  = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(300) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname, @pred nvarchar(max);
DECLARE @obj nvarchar(600), @label nvarchar(300), @sql nvarchar(max), @cnt bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name, p.pred
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    OUTER APPLY (
        SELECT STRING_AGG(CAST(QUOTENAME(c.name) + N' IN (@full, @email, @phone)' AS nvarchar(max)), N' OR ') AS pred
        FROM sys.columns AS c
        JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
        WHERE c.object_id = t.object_id
          AND ty.name IN (N'nvarchar', N'varchar', N'nchar', N'char')
    ) AS p
    WHERE t.is_ms_shipped = 0;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table, @pred;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @label = @schema + N'.' + @table;
    SET @cnt = 0;

    IF @pred IS NOT NULL
    BEGIN
        SET @obj = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
        SET @sql = N'SELECT @cnt_out = COUNT(*) FROM ' + @obj + N' WHERE ' + @pred + N';';

        EXEC sp_executesql
            @sql,
            N'@full nvarchar(200), @email nvarchar(200), @phone nvarchar(50), @cnt_out bigint OUTPUT',
            @full = @full,
            @email = @email,
            @phone = @phone,
            @cnt_out = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found) VALUES (@label, @cnt);

    FETCH NEXT FROM tbl_cur INTO @schema, @table, @pred;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
