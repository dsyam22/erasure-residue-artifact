```sql
SET NOCOUNT ON;

DECLARE @vals TABLE (v nvarchar(200) NOT NULL);
INSERT INTO @vals (v) VALUES
    (N'Subject_0000042'),
    (N'subject42@example.invalid'),
    (N'+1-555-0000042');

CREATE TABLE #results (
    object_name nvarchar(258) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname, @oid int, @pred nvarchar(max), @n bigint, @sql nvarchar(max);

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name, t.object_id
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table, @oid;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @n = 0;

    -- Build a predicate over all character columns (varchar, nvarchar, char, nchar)
    -- against every subject value. text/ntext/xml/varbinary are not searched here.
    SELECT @pred = STRING_AGG(
        CAST(N'(CHARINDEX(N''' + REPLACE(v.v, N'''', N'''''') + N''', '
             + QUOTENAME(c.name) + N' COLLATE Latin1_General_CI_AS) > 0)' AS nvarchar(max)),
        N' OR ')
    FROM sys.columns AS c
    CROSS JOIN @vals AS v
    WHERE c.object_id = @oid
      AND c.system_type_id IN (167, 231, 175, 239);  -- varchar, nvarchar, char, nchar

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @n_out = COUNT(*) FROM ' + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';
        EXEC sys.sp_executesql @sql, N'@n_out bigint OUTPUT', @n_out = @n OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (CAST(@schema + N'.' + @table AS nvarchar(258)), ISNULL(@n, 0));

    FETCH NEXT FROM tbl_cur INTO @schema, @table, @oid;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
