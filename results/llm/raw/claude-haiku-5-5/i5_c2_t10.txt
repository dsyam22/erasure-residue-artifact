```sql
SET NOCOUNT ON;

DECLARE @vals TABLE (v nvarchar(200) NOT NULL);
INSERT INTO @vals (v) VALUES (N'Subject_0000042'), (N'subject42@example.invalid'), (N'+1-555-0000042');

CREATE TABLE #res (
    object_name nvarchar(300) NOT NULL PRIMARY KEY,
    rows_found  bigint NOT NULL
);

-- Register every user table (includes dbo.customer and dbo.customer_history) with 0 by default
INSERT INTO #res (object_name, rows_found)
SELECT QUOTENAME(s.name) + N'.' + QUOTENAME(t.name), 0
FROM sys.tables AS t
JOIN sys.schemas AS s ON s.schema_id = t.schema_id
WHERE t.is_ms_shipped = 0;

DECLARE @oid int, @fq nvarchar(300), @pred nvarchar(max), @cnt bigint, @sql nvarchar(max);

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT t.object_id, QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @oid, @fq;

WHILE @@FETCH_STATUS = 0
BEGIN
    -- Predicate: any character column contains any of the subject's personal data values
    SET @pred = (
        SELECT STRING_AGG(CAST(
            N'CHARINDEX(N''' + REPLACE(v.v, N'''', N'''''') + N''', CAST(' + QUOTENAME(c.name)
            + N' AS nvarchar(max))) > 0' AS nvarchar(max)), N' OR ')
        FROM sys.columns AS c
        CROSS JOIN @vals AS v
        WHERE c.object_id = @oid
          AND c.system_type_id IN (167, 175, 231, 239) -- varchar, char, nvarchar, nchar
    );

    SET @cnt = 0;

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @c = COUNT_BIG(*) FROM ' + @fq + N' WHERE ' + @pred + N';';
        EXEC sys.sp_executesql @sql, N'@c bigint OUTPUT', @c = @cnt OUTPUT;
    END

    UPDATE #res SET rows_found = @cnt WHERE object_name = @fq;

    FETCH NEXT FROM tbl_cur INTO @oid, @fq;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #res
ORDER BY object_name;
```
