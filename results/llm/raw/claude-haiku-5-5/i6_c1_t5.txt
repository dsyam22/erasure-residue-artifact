```sql
SET NOCOUNT ON;

CREATE TABLE #r
(
    object_name nvarchar(300) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname;
DECLARE @obj_display nvarchar(300), @obj_quoted nvarchar(400);
DECLARE @pred nvarchar(max), @sql nvarchar(max);

DECLARE tbl CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0;

OPEN tbl;
FETCH NEXT FROM tbl INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj_display = @schema + N'.' + @table;
    SET @obj_quoted  = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @pred = NULL;

    SELECT @pred = STRING_AGG(
               CONVERT(nvarchar(max),
                   N'CONVERT(nvarchar(max),' + QUOTENAME(c.name) + N') LIKE N''%+1-555-0000042%'''
                 + N' OR CONVERT(nvarchar(max),' + QUOTENAME(c.name) + N') LIKE N''%5550000042%'''),
               N' OR ')
    FROM sys.columns AS c
    JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
    WHERE c.object_id = OBJECT_ID(@obj_quoted)
      AND ty.name IN (N'varchar', N'nvarchar', N'char', N'nchar', N'xml');

    IF @pred IS NULL
        SET @pred = N'1 = 0';

    SET @sql = N'INSERT INTO #r (object_name, rows_found) '
             + N'SELECT @o, COUNT_BIG(*) FROM ' + @obj_quoted
             + N' WHERE ' + @pred + N';';

    EXEC sys.sp_executesql @sql, N'@o nvarchar(300)', @o = @obj_display;

    FETCH NEXT FROM tbl INTO @schema, @table;
END

CLOSE tbl;
DEALLOCATE tbl;

SELECT object_name, rows_found
FROM #r
ORDER BY object_name;

DROP TABLE #r;
```
