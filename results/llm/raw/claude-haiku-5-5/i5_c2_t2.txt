```sql
SET NOCOUNT ON;

DECLARE @p1 nvarchar(200) = N'Subject_0000042';
DECLARE @p2 nvarchar(200) = N'subject42@example.invalid';
DECLARE @p3 nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    -- Build a predicate over all character-type columns of this table
    SELECT @pred = STRING_AGG(
               CAST(N'('
                  + N'CHARINDEX(@p1, ' + QUOTENAME(c.name) + N') > 0 OR '
                  + N'CHARINDEX(@p2, ' + QUOTENAME(c.name) + N') > 0 OR '
                  + N'CHARINDEX(@p3, ' + QUOTENAME(c.name) + N') > 0)' AS nvarchar(max)),
               N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = OBJECT_ID(QUOTENAME(@schema) + N'.' + QUOTENAME(@table))
      AND c.system_type_id IN (167, 175, 231, 239);  -- varchar, char, nvarchar, nchar

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @cnt_out = COUNT(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';

        EXEC sp_executesql
             @sql,
             N'@p1 nvarchar(200), @p2 nvarchar(200), @p3 nvarchar(200), @cnt_out bigint OUTPUT',
             @p1 = @p1, @p2 = @p2, @p3 = @p3, @cnt_out = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (CAST(@schema + N'.' + @table AS sysname), ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
