```sql
SET NOCOUNT ON;

CREATE TABLE #hits
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @object_id int,
        @schema    sysname,
        @table     sysname,
        @pred      nvarchar(max),
        @sql       nvarchar(max),
        @cnt       bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT t.object_id, s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @object_id, @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    -- Build predicate: any character column equal to any of the subject's
    -- original full_name, email or phone. (customer_id is a key, not matched by value.)
    SET @pred = NULL;
    SELECT @pred = STRING_AGG(
               CAST(
                 QUOTENAME(c.name) + N' IN (N''Subject_0000042'', N''subject42@example.invalid'', N''+1-555-0000042'')'
                 AS nvarchar(max)),
               N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = @object_id
      AND c.system_type_id IN (167, 175, 231, 239); -- varchar, char, nvarchar, nchar

    IF @pred IS NULL
    BEGIN
        SET @cnt = 0;
    END
    ELSE
    BEGIN
        SET @sql = N'SELECT @n = COUNT_BIG(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';
        EXEC sys.sp_executesql
             @sql,
             N'@n bigint OUTPUT',
             @n = @cnt OUTPUT;
    END

    INSERT INTO #hits (object_name, rows_found)
    VALUES (@schema + N'.' + @table, ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl_cur INTO @object_id, @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #hits
ORDER BY object_name;
```
