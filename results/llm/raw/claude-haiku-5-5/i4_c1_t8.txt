```sql
SET NOCOUNT ON;

DECLARE @subjName nvarchar(100) = N'Subject_0000042',
        @subjEmail nvarchar(200) = N'subject42@example.invalid',
        @subjPhone nvarchar(50) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name sysname NOT NULL,
    rows_found  int     NOT NULL
);

DECLARE @sch sysname, @tb sysname, @pred nvarchar(max), @sql nvarchar(max), @cnt int;

DECLARE tbl CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    ORDER BY s.name, t.name;

OPEN tbl;
FETCH NEXT FROM tbl INTO @sch, @tb;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    SELECT @pred = STRING_AGG(CAST(
            N'CHARINDEX(@pName, ' + QUOTENAME(c.name) + N') > 0 OR '
          + N'CHARINDEX(@pEmail, ' + QUOTENAME(c.name) + N') > 0 OR '
          + N'CHARINDEX(@pPhone, ' + QUOTENAME(c.name) + N') > 0'
          AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = OBJECT_ID(QUOTENAME(@sch) + N'.' + QUOTENAME(@tb))
      AND c.system_type_id IN (167, 175, 231, 239);  -- varchar, char, nvarchar, nchar

    SET @sql = N'SELECT @pCnt = COUNT(*) FROM ' + QUOTENAME(@sch) + N'.' + QUOTENAME(@tb)
             + CASE WHEN @pred IS NULL THEN N' WHERE 1 = 0'
                    ELSE N' WHERE ' + @pred END;

    EXEC sp_executesql
         @sql,
         N'@pName nvarchar(100), @pEmail nvarchar(200), @pPhone nvarchar(50), @pCnt int OUTPUT',
         @pName = @subjName,
         @pEmail = @subjEmail,
         @pPhone = @subjPhone,
         @pCnt = @cnt OUTPUT;

    INSERT INTO #results (object_name, rows_found)
    VALUES (@sch + N'.' + @tb, @cnt);

    FETCH NEXT FROM tbl INTO @sch, @tb;
END;

CLOSE tbl;
DEALLOCATE tbl;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
