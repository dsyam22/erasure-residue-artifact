```sql
SET NOCOUNT ON;

DECLARE @p1 nvarchar(64) = N'%+1-555-0000042%';
DECLARE @p2 nvarchar(64) = N'%15550000042%';

CREATE TABLE #checked
(
    object_name nvarchar(300) NOT NULL PRIMARY KEY,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname, @oid int,
        @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tcur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name, t.object_id
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0;

OPEN tcur;
FETCH NEXT FROM tcur INTO @schema, @table, @oid;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    SELECT @pred = STRING_AGG(
               CAST(N'CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max)) LIKE @p1 OR CAST('
                    + QUOTENAME(c.name) + N' AS nvarchar(max)) LIKE @p2' AS nvarchar(max)),
               N' OR ')
    FROM sys.columns AS c
    JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
    WHERE c.object_id = @oid
      AND ty.name IN (N'char', N'varchar', N'nchar', N'nvarchar', N'text', N'ntext', N'xml');

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @n = COUNT(*) FROM ' + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';
        EXEC sp_executesql @sql,
             N'@n bigint OUTPUT, @p1 nvarchar(64), @p2 nvarchar(64)',
             @n = @cnt OUTPUT, @p1 = @p1, @p2 = @p2;
    END

    INSERT INTO #checked (object_name, rows_found)
    VALUES (@schema + N'.' + @table, @cnt);

    FETCH NEXT FROM tcur INTO @schema, @table, @oid;
END

CLOSE tcur;
DEALLOCATE tcur;

SELECT object_name, rows_found
FROM #checked
ORDER BY object_name;
```
