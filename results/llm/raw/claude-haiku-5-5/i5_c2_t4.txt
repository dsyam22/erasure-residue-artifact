```sql
SET NOCOUNT ON;

CREATE TABLE #results
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @obj nvarchar(300), @pred nvarchar(max), @sql nvarchar(max);

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @pred = NULL;

    SELECT @pred = STRING_AGG(CAST(
        N'(CHARINDEX(@p1, CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0 OR '
      + N'CHARINDEX(@p2, CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0 OR '
      + N'CHARINDEX(@p3, CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0)' AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
    WHERE c.object_id = OBJECT_ID(@obj)
      AND ty.name IN (N'char', N'varchar', N'nchar', N'nvarchar');

    IF @pred IS NULL
        SET @pred = N'1 = 0';

    SET @sql = N'INSERT INTO #results (object_name, rows_found) '
             + N'SELECT N''' + REPLACE(@schema, N'''', N'''''') + N'.' + REPLACE(@table, N'''', N'''''') + N''', COUNT_BIG(*) '
             + N'FROM ' + @obj + N' WHERE ' + @pred + N';';

    EXEC sp_executesql
        @sql,
        N'@p1 nvarchar(200), @p2 nvarchar(200), @p3 nvarchar(200)',
        @p1 = N'Subject_0000042',
        @p2 = N'subject42@example.invalid',
        @p3 = N'+1-555-0000042';

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
