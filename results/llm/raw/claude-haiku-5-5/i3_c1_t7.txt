```sql
SET NOCOUNT ON;

DECLARE @fn nvarchar(200) = N'subject_0000042';
DECLARE @em nvarchar(200) = N'subject42@example.invalid';
DECLARE @ph nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(300) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname, @cond nvarchar(max), @sql nvarchar(max), @n bigint;

DECLARE tbl CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0;

OPEN tbl;
FETCH NEXT FROM tbl INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @cond = NULL;
    SET @n = 0;

    -- Build a predicate over all character columns of the current table
    SELECT @cond = STRING_AGG(
               CAST(N'LOWER(LTRIM(RTRIM(' + QUOTENAME(c.name) + N'))) IN (@fn, @em, @ph)' AS nvarchar(max)),
               N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = OBJECT_ID(QUOTENAME(@schema) + N'.' + QUOTENAME(@table))
      AND c.system_type_id IN (167, 175, 231, 239);  -- varchar, char, nvarchar, nchar

    IF @cond IS NULL
    BEGIN
        INSERT INTO #results (object_name, rows_found)
        VALUES (@schema + N'.' + @table, 0);
    END
    ELSE
    BEGIN
        SET @sql = N'SELECT @n_out = COUNT(*) FROM ' + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @cond + N';';

        EXEC sp_executesql
             @sql,
             N'@fn nvarchar(200), @em nvarchar(200), @ph nvarchar(200), @n_out bigint OUTPUT',
             @fn = @fn, @em = @em, @ph = @ph, @n_out = @n OUTPUT;

        INSERT INTO #results (object_name, rows_found)
        VALUES (@schema + N'.' + @table, @n);
    END

    FETCH NEXT FROM tbl INTO @schema, @table;
END

CLOSE tbl;
DEALLOCATE tbl;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
