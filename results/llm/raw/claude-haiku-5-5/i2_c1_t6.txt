```sql
SET NOCOUNT ON;

DECLARE @p1 nvarchar(256) = N'Subject_0000042';
DECLARE @p2 nvarchar(256) = N'subject42@example.invalid';
DECLARE @p3 nvarchar(256) = N'+1-555-0000042';

CREATE TABLE #hits (
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT TABLE_SCHEMA, TABLE_NAME
    FROM INFORMATION_SCHEMA.TABLES
    WHERE TABLE_TYPE = 'BASE TABLE'
    ORDER BY TABLE_SCHEMA, TABLE_NAME;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    SELECT @pred = STRING_AGG(
               CAST(QUOTENAME(c.COLUMN_NAME) + N' IN (@p1, @p2, @p3)' AS nvarchar(max)),
               N' OR ')
    FROM INFORMATION_SCHEMA.COLUMNS AS c
    WHERE c.TABLE_SCHEMA = @schema
      AND c.TABLE_NAME   = @table
      AND c.DATA_TYPE IN ('char', 'varchar', 'nchar', 'nvarchar');

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @c = COUNT(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';

        EXEC sp_executesql
            @sql,
            N'@p1 nvarchar(256), @p2 nvarchar(256), @p3 nvarchar(256), @c bigint OUTPUT',
            @p1 = @p1, @p2 = @p2, @p3 = @p3, @c = @cnt OUTPUT;
    END

    INSERT INTO #hits (object_name, rows_found)
    VALUES (CAST(@schema + N'.' + @table AS sysname), ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #hits
ORDER BY object_name;

DROP TABLE #hits;
```
