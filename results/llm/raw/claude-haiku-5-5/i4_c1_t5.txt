```sql
SET NOCOUNT ON;

DECLARE @p1 nvarchar(200) = N'Subject_0000042',
        @p2 nvarchar(200) = N'subject42@example.invalid',
        @p3 nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name nvarchar(400) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @schema sysname, @table sysname,
        @obj nvarchar(400), @cond nvarchar(max),
        @sql nvarchar(max), @n bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT SCHEMA_NAME(t.schema_id), t.name
    FROM sys.tables AS t
    ORDER BY SCHEMA_NAME(t.schema_id), t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj  = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @cond = NULL;
    SET @n    = 0;

    SELECT @cond = STRING_AGG(
               CAST(N'(CHARINDEX(@p1, ' + x.e + N') > 0 OR CHARINDEX(@p2, ' + x.e
                    + N') > 0 OR CHARINDEX(@p3, ' + x.e + N') > 0)' AS nvarchar(max)),
               N' OR ')
    FROM (
        SELECT CASE WHEN ty.name = N'xml'
                    THEN N'CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))'
                    ELSE QUOTENAME(c.name)
               END AS e
        FROM sys.columns AS c
        JOIN sys.types   AS ty ON ty.user_type_id = c.user_type_id
        WHERE c.object_id = OBJECT_ID(@obj)
          AND ty.name IN (N'nvarchar', N'varchar', N'nchar', N'char', N'xml')
    ) AS x;

    IF @cond IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @n = COUNT_BIG(*) FROM ' + @obj + N' WHERE ' + @cond;

        EXEC sp_executesql
            @sql,
            N'@p1 nvarchar(200), @p2 nvarchar(200), @p3 nvarchar(200), @n bigint OUTPUT',
            @p1 = @p1, @p2 = @p2, @p3 = @p3, @n = @n OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (@schema + N'.' + @table, @n);

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
