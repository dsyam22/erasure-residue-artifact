```sql
SET NOCOUNT ON;

DECLARE @cid int = 42,
        @name nvarchar(200) = N'Subject_0000042',
        @email nvarchar(400) = N'subject42@example.invalid',
        @phone nvarchar(60) = N'+1-555-0000042';

SELECT
    CAST(N'dbo.customer' AS nvarchar(256)) AS object_name,
    (SELECT COUNT(*)
     FROM dbo.customer c
     WHERE c.customer_id = @cid
        OR c.full_name = @name
        OR c.email = @email
        OR c.phone = @phone) AS rows_found
UNION ALL
SELECT
    CAST(N'cdc.dbo_customer_CT' AS nvarchar(256)),
    (SELECT COUNT(*)
     FROM cdc.dbo_customer_CT x
     WHERE x.customer_id = @cid
        OR x.full_name = @name
        OR x.email = @email
        OR x.phone = @phone);
```
