```sql
-- SCRIPT
SET NOCOUNT ON;

SELECT object_name, rows_found
FROM (
    SELECT CAST('dbo.customer' AS nvarchar(256)) AS object_name,
           (SELECT COUNT(*)
            FROM dbo.customer AS x
            WHERE x.customer_id = 42
               OR x.full_name = N'Subject_0000042'
               OR x.email = N'subject42@example.invalid'
               OR x.phone = N'+1-555-0000042') AS rows_found
    UNION ALL
    SELECT CAST('reporting.customer_replica' AS nvarchar(256)),
           (SELECT COUNT(*)
            FROM reporting.customer_replica AS x
            WHERE x.customer_id = 42
               OR x.full_name = N'Subject_0000042'
               OR x.email = N'subject42@example.invalid'
               OR x.phone = N'+1-555-0000042')
) AS checks
ORDER BY object_name;
```
