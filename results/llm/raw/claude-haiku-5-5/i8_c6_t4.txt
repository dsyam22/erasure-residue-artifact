SET NOCOUNT ON;

SELECT
    CAST('dbo.customer' AS nvarchar(256)) AS object_name,
    CAST(COUNT_BIG(*) AS bigint) AS rows_found
FROM dbo.customer
WHERE customer_id = 42
   OR full_name = N'Subject_0000042'
   OR email = N'subject42@example.invalid'
   OR phone = N'+1-555-0000042';
