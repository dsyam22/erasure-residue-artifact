```sql
SET NOCOUNT ON;

CREATE TABLE #results (
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @object_id int;
DECLARE @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name, t.object_id
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table, @object_id;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    -- Build a predicate over all character-type columns of this table
    SELECT @pred = STRING_AGG(CAST(N'(' + p.expr + N')' AS nvarchar(max)), N' OR ')
    FROM (
        SELECT
            N'(CHARINDEX(N''Subject_0000042'', CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0 OR '
          + N'CHARINDEX(N''subject42@example.invalid'', CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0 OR '
          + N'CHARINDEX(N''+1-555-0000042'', CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0)' AS expr
        FROM sys.columns AS c
        JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
        WHERE c.object_id = @object_id
          AND ty.name IN (N'char', N'varchar', N'nchar', N'nvarchar')
    ) AS p;

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @cnt_out = COUNT_BIG(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';

        EXEC sys.sp_executesql
            @sql,
            N'@cnt_out bigint OUTPUT',
            @cnt_out = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (QUOTENAME(@schema) + N'.' + QUOTENAME(@table), ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl_cur INTO @schema, @table, @object_id;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
