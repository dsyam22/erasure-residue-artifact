```sql
SET NOCOUNT ON;

DECLARE @n nvarchar(200) = N'Subject_0000042';
DECLARE @e nvarchar(200) = N'subject42@example.invalid';
DECLARE @p nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @s sysname, @t sysname, @oid int;
DECLARE @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tcur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name, t.object_id
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id;

OPEN tcur;
FETCH NEXT FROM tcur INTO @s, @t, @oid;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    SELECT @pred = STRING_AGG(
            CAST(
                CASE WHEN ty.name = N'xml'
                     THEN N'CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))'
                     ELSE QUOTENAME(c.name)
                END
                + N' LIKE N''%'' + @n + N''%'''
            AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
    WHERE c.object_id = @oid
      AND ty.name IN (N'nvarchar', N'varchar', N'nchar', N'char', N'xml');

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @cnt = COUNT(*) FROM '
                 + QUOTENAME(@s) + N'.' + QUOTENAME(@t)
                 + N' WHERE ' + @pred + N';';

        EXEC sp_executesql
            @sql,
            N'@n nvarchar(200), @e nvarchar(200), @p nvarchar(200), @cnt bigint OUTPUT',
            @n = @n, @e = @e, @p = @p, @cnt = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (@s + N'.' + @t, @cnt);

    FETCH NEXT FROM tcur INTO @s, @t, @oid;
END

CLOSE tcur;
DEALLOCATE tcur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
