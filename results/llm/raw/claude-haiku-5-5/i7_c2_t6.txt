```sql
SET NOCOUNT ON;

DECLARE @v1 nvarchar(200) = N'Subject_0000042',
        @v2 nvarchar(200) = N'subject42@example.invalid',
        @v3 nvarchar(200) = N'+1-555-0000042';

CREATE TABLE #results (
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @obj nvarchar(300),
        @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @pred = NULL;
    SET @cnt = 0;

    -- Build predicate: any character-type column equal to any of the subject's original PII values
    SELECT @pred = STRING_AGG(
               CAST(QUOTENAME(c.name) + N' IN (@v1, @v2, @v3)' AS nvarchar(max)),
               N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = OBJECT_ID(@obj)
      AND c.system_type_id IN (167, 175, 231, 239);  -- varchar, char, nvarchar, nchar

    IF @pred IS NULL
        SET @pred = N'1 = 0';

    SET @sql = N'SELECT @c = COUNT(*) FROM ' + @obj + N' WHERE ' + @pred + N';';

    EXEC sys.sp_executesql
         @sql,
         N'@v1 nvarchar(200), @v2 nvarchar(200), @v3 nvarchar(200), @c bigint OUTPUT',
         @v1 = @v1, @v2 = @v2, @v3 = @v3, @c = @cnt OUTPUT;

    INSERT INTO #results (object_name, rows_found)
    VALUES (CAST(@schema + N'.' + @table AS sysname), @cnt);

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
