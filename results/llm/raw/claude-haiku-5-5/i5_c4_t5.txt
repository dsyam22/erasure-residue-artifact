```sql
SET NOCOUNT ON;

CREATE TABLE #results
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @needles TABLE (v nvarchar(200) NOT NULL);
INSERT INTO @needles (v)
VALUES (N'Subject_0000042'),
       (N'subject42@example.invalid'),
       (N'+1-555-0000042');

DECLARE @object_id int,
        @schema    sysname,
        @table     sysname,
        @where     nvarchar(max),
        @sql       nvarchar(max),
        @cnt       bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT t.object_id, s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @object_id, @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @where = NULL;

    SELECT @where = STRING_AGG(
               CAST(N'(CHARINDEX(N''' + REPLACE(n.v, N'''', N'''''') + N''', CAST('
                    + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0)' AS nvarchar(max)),
               N' OR ')
           )
    FROM sys.columns AS c
    CROSS JOIN @needles AS n
    WHERE c.object_id = @object_id
      AND TYPE_NAME(c.user_type_id) IN (N'char', N'varchar', N'nchar', N'nvarchar',
                                        N'xml', N'text', N'ntext');

    SET @cnt = 0;

    IF @where IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @c = COUNT(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @where + N';';
        EXEC sp_executesql @sql, N'@c bigint OUTPUT', @c = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (@schema + N'.' + @table, @cnt);

    FETCH NEXT FROM tbl_cur INTO @object_id, @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;
```
