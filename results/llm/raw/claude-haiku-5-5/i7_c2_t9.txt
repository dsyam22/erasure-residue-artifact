```sql
SET NOCOUNT ON;

CREATE TABLE #results
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @obj nvarchar(400), @pred nvarchar(max),
        @sql nvarchar(max), @cnt bigint;

DECLARE tbl CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl;
FETCH NEXT FROM tbl INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj  = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @pred = NULL;
    SET @cnt  = 0;

    -- Build predicate: any character column equal to one of the subject's original values,
    -- or any customer_id column equal to 42.
    SELECT @pred = STRING_AGG(CAST(x.p AS nvarchar(max)), N' OR ')
    FROM
    (
        SELECT QUOTENAME(c.name) + N' IN (N''Subject_0000042'', N''subject42@example.invalid'', N''+1-555-0000042'')' AS p
        FROM sys.columns AS c
        WHERE c.object_id = OBJECT_ID(@obj)
          AND c.system_type_id IN (167, 175, 231, 239) -- varchar, char, nvarchar, nchar
        UNION ALL
        SELECT QUOTENAME(c.name) + N' = 42'
        FROM sys.columns AS c
        WHERE c.object_id = OBJECT_ID(@obj)
          AND c.name = N'customer_id'
          AND c.system_type_id = 56 -- int
    ) AS x;

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @cnt_out = COUNT_BIG(*) FROM ' + @obj + N' WHERE ' + @pred + N';';
        EXEC sys.sp_executesql @sql, N'@cnt_out bigint OUTPUT', @cnt_out = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (@schema + N'.' + @table, ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl INTO @schema, @table;
END

CLOSE tbl;
DEALLOCATE tbl;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
