```sql
SET NOCOUNT ON;

SELECT v.object_name,
       v.rows_found
FROM (
    SELECT N'dbo.customer' AS object_name,
           (SELECT COUNT(*)
            FROM dbo.customer AS c
            WHERE c.customer_id = 42
               OR c.full_name = N'Subject_0000042'
               OR c.email = N'subject42@example.invalid'
               OR c.phone = N'+1-555-0000042') AS rows_found
    UNION ALL
    SELECT N'dbo.customer_history',
           (SELECT COUNT(*)
            FROM dbo.customer_history AS h
            WHERE h.customer_id = 42
               OR h.full_name = N'Subject_0000042'
               OR h.email = N'subject42@example.invalid'
               OR h.phone = N'+1-555-0000042')
    UNION ALL
    SELECT N'sys.sql_modules',
           (SELECT COUNT(*)
            FROM sys.sql_modules AS m
            WHERE m.definition LIKE N'%Subject[_]0000042%'
               OR m.definition LIKE N'%subject42@example%'
               OR m.definition LIKE N'%555-0000042%'
               OR m.definition LIKE N'%customer_id = 42%')
) AS v
ORDER BY v.object_name;
```
