```sql
SET NOCOUNT ON;

DECLARE @v1 nvarchar(256) = N'Subject_0000042',
        @v2 nvarchar(256) = N'subject42@example.invalid',
        @v3 nvarchar(256) = N'+1-555-0000042';

CREATE TABLE #scan
(
    object_name nvarchar(300) NOT NULL PRIMARY KEY,
    rows_found  bigint        NOT NULL
);

DECLARE @display nvarchar(300),
        @full    nvarchar(300),
        @pred    nvarchar(max),
        @sql     nvarchar(max),
        @cnt     bigint;

DECLARE tcur CURSOR LOCAL FAST_FORWARD FOR
    SELECT
        SCHEMA_NAME(t.schema_id) + N'.' + t.name,
        QUOTENAME(SCHEMA_NAME(t.schema_id)) + N'.' + QUOTENAME(t.name),
        (
            SELECT STRING_AGG(CAST(N'c.' + QUOTENAME(col.name) + N' IN (@v1, @v2, @v3)' AS nvarchar(max)), N' OR ')
            FROM sys.columns AS col
            JOIN sys.types AS ty
              ON ty.user_type_id = col.user_type_id
            WHERE col.object_id = t.object_id
              AND ty.name IN (N'nvarchar', N'varchar', N'nchar', N'char')
        )
    FROM sys.tables AS t
    WHERE t.is_ms_shipped = 0;

OPEN tcur;
FETCH NEXT FROM tcur INTO @display, @full, @pred;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @cnt = 0;

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @n = COUNT_BIG(*) FROM ' + @full + N' AS c WHERE ' + @pred + N';';
        EXEC sys.sp_executesql
             @sql,
             N'@v1 nvarchar(256), @v2 nvarchar(256), @v3 nvarchar(256), @n bigint OUTPUT',
             @v1 = @v1, @v2 = @v2, @v3 = @v3, @n = @cnt OUTPUT;
    END

    INSERT INTO #scan (object_name, rows_found) VALUES (@display, @cnt);

    FETCH NEXT FROM tcur INTO @display, @full, @pred;
END

CLOSE tcur;
DEALLOCATE tcur;

SELECT object_name, rows_found
FROM #scan
ORDER BY object_name;

DROP TABLE #scan;
```
