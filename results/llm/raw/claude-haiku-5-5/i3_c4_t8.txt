```sql
SET NOCOUNT ON;

CREATE TABLE #results
(
    object_name nvarchar(400) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE
    @schema nvarchar(128),
    @table  nvarchar(128),
    @obj    nvarchar(400),
    @pred   nvarchar(max),
    @sql    nvarchar(max),
    @cnt    bigint,
    @p1     nvarchar(200) = N'Subject_0000042',
    @p2     nvarchar(200) = N'subject42@example.invalid',
    @p3     nvarchar(200) = N'+1-555-0000042';

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj  = @schema + N'.' + @table;
    SET @pred = NULL;
    SET @cnt  = 0;

    SELECT @pred = STRING_AGG(CAST(
        N'(CHARINDEX(@p1, CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0'
      + N' OR CHARINDEX(@p2, CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0'
      + N' OR CHARINDEX(@p3, CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max))) > 0)'
      AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
    WHERE c.object_id = OBJECT_ID(QUOTENAME(@schema) + N'.' + QUOTENAME(@table))
      AND ty.name IN (N'char', N'varchar', N'nchar', N'nvarchar', N'text', N'ntext', N'xml', N'sysname');

    IF @pred IS NULL
    BEGIN
        INSERT INTO #results (object_name, rows_found) VALUES (@obj, 0);
    END
    ELSE
    BEGIN
        SET @sql = N'SELECT @cnt = COUNT_BIG(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';

        EXEC sp_executesql
            @sql,
            N'@p1 nvarchar(200), @p2 nvarchar(200), @p3 nvarchar(200), @cnt bigint OUTPUT',
            @p1 = @p1, @p2 = @p2, @p3 = @p3, @cnt = @cnt OUTPUT;

        INSERT INTO #results (object_name, rows_found) VALUES (@obj, ISNULL(@cnt, 0));
    END

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
