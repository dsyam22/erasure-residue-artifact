```sql
SET NOCOUNT ON;

DECLARE @v1 nvarchar(200) = N'Subject_0000042';
DECLARE @v2 nvarchar(200) = N'subject42@example.invalid';
DECLARE @v3 nvarchar(200) = N'+1-555-0000042';
DECLARE @v4 nvarchar(200) = N'Subject_0000042';

CREATE TABLE #scan_results
(
    object_name nvarchar(260) NOT NULL,
    rows_found  bigint        NOT NULL
);

DECLARE @oid int, @obj nvarchar(260), @cond nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT t.object_id, QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @oid, @obj;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @cond = NULL;
    SET @cnt = 0;

    -- Build a predicate over all character columns of this table.
    SELECT @cond = STRING_AGG(CAST(
        N'(CHARINDEX(@v1, ' + QUOTENAME(c.name) + N') > 0 OR '
      + N'CHARINDEX(@v2, ' + QUOTENAME(c.name) + N') > 0 OR '
      + N'CHARINDEX(@v3, ' + QUOTENAME(c.name) + N') > 0 OR '
      + N'CHARINDEX(@v4, ' + QUOTENAME(c.name) + N') > 0)'
      AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = @oid
      AND c.system_type_id IN (167, 231, 175, 239);  -- varchar, nvarchar, char, nchar

    IF @cond IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @cnt_out = COUNT_BIG(*) FROM ' + @obj + N' WHERE ' + @cond + N';';

        EXEC sys.sp_executesql
            @sql,
            N'@v1 nvarchar(200), @v2 nvarchar(200), @v3 nvarchar(200), @v4 nvarchar(200), @cnt_out bigint OUTPUT',
            @v1 = @v1, @v2 = @v2, @v3 = @v3, @v4 = @v4,
            @cnt_out = @cnt OUTPUT;
    END

    INSERT INTO #scan_results (object_name, rows_found)
    VALUES (@obj, ISNULL(@cnt, 0));

    FETCH NEXT FROM tbl_cur INTO @oid, @obj;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #scan_results
ORDER BY object_name;

DROP TABLE #scan_results;
```
