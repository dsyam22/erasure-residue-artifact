```sql
SET NOCOUNT ON;

DECLARE @full_name nvarchar(200) = N'Subject_0000042';
DECLARE @email     nvarchar(320) = N'subject42@example.invalid';
DECLARE @phone     nvarchar(50)  = N'+1-555-0000042';

SELECT v.object_name,
       v.rows_found
FROM (
    SELECT N'dbo.customer' AS object_name,
           (SELECT COUNT(*)
              FROM dbo.customer AS c
             WHERE c.full_name = @full_name
                OR c.email     = @email
                OR c.phone     = @phone) AS rows_found
    UNION ALL
    SELECT N'reporting.customer_replica' AS object_name,
           (SELECT COUNT(*)
              FROM reporting.customer_replica AS r
             WHERE r.full_name = @full_name
                OR r.email     = @email
                OR r.phone     = @phone) AS rows_found
) AS v;
```
