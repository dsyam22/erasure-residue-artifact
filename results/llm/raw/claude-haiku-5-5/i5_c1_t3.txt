```sql
SET NOCOUNT ON;

CREATE TABLE #erasure_check
(
    object_name sysname NOT NULL,
    rows_found  bigint  NOT NULL
);

DECLARE @schema sysname, @table sysname, @pred nvarchar(max), @sql nvarchar(max), @n bigint;

DECLARE tbl_cur CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl_cur;
FETCH NEXT FROM tbl_cur INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;

    -- Build a predicate over all character columns: name, email, phone patterns.
    -- '_' is escaped as [_] so LIKE treats it literally.
    SELECT @pred = STRING_AGG(
        CAST(
            N'(' + QUOTENAME(c.name) + N' LIKE N''%Subject[_]0000042%'''
          + N' OR ' + QUOTENAME(c.name) + N' LIKE N''%subject42@example.invalid%'''
          + N' OR ' + QUOTENAME(c.name) + N' LIKE N''%+1-555-0000042%'')'
        AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    WHERE c.object_id = OBJECT_ID(QUOTENAME(@schema) + N'.' + QUOTENAME(@table))
      AND c.system_type_id IN (167, 175, 231, 239);  -- varchar, char, nvarchar, nchar

    -- For the customer table, also check for any remaining row with the erased customer_id.
    IF @schema = N'dbo' AND @table = N'customer'
    BEGIN
        SET @pred = COALESCE(@pred + N' OR ', N'') + N'([customer_id] = 42)';
    END

    IF @pred IS NULL
    BEGIN
        INSERT #erasure_check (object_name, rows_found)
        VALUES (@schema + N'.' + @table, 0);
    END
    ELSE
    BEGIN
        SET @sql = N'SELECT @n = COUNT_BIG(*) FROM '
                 + QUOTENAME(@schema) + N'.' + QUOTENAME(@table)
                 + N' WHERE ' + @pred + N';';

        SET @n = 0;
        EXEC sp_executesql @sql, N'@n bigint OUTPUT', @n = @n OUTPUT;

        INSERT #erasure_check (object_name, rows_found)
        VALUES (@schema + N'.' + @table, @n);
    END

    FETCH NEXT FROM tbl_cur INTO @schema, @table;
END

CLOSE tbl_cur;
DEALLOCATE tbl_cur;

SELECT object_name, rows_found
FROM #erasure_check
ORDER BY object_name;

DROP TABLE #erasure_check;
```
