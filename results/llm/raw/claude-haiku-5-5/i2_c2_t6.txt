```sql
SET NOCOUNT ON;

DECLARE @fn nvarchar(200) = N'Subject_0000042';
DECLARE @em nvarchar(320) = N'subject42@example.invalid';
DECLARE @ph nvarchar(50)  = N'+1-555-0000042';

CREATE TABLE #targets
(
    object_id   int       NOT NULL PRIMARY KEY,
    schema_name sysname   NOT NULL,
    table_name  sysname   NOT NULL
);

CREATE TABLE #results
(
    object_name nvarchar(300) NOT NULL,
    rows_found  bigint        NOT NULL
);

INSERT INTO #targets (object_id, schema_name, table_name)
SELECT t.object_id, s.name, t.name
FROM sys.tables AS t
JOIN sys.schemas AS s ON s.schema_id = t.schema_id
WHERE t.is_ms_shipped = 0;

DECLARE @oid int, @s sysname, @t sysname;
DECLARE @pred nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE c CURSOR LOCAL FAST_FORWARD FOR
    SELECT object_id, schema_name, table_name FROM #targets ORDER BY schema_name, table_name;

OPEN c;
FETCH NEXT FROM c INTO @oid, @s, @t;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @pred = NULL;
    SET @cnt = 0;

    SELECT @pred = STRING_AGG(
               CAST(QUOTENAME(col.name) + N' IN (@fn, @em, @ph)' AS nvarchar(max)),
               N' OR ')
    FROM sys.columns AS col
    JOIN sys.types AS ty ON ty.user_type_id = col.user_type_id
    WHERE col.object_id = @oid
      AND ty.name IN (N'nvarchar', N'varchar', N'nchar', N'char');

    IF @pred IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @n = COUNT_BIG(*) FROM '
                 + QUOTENAME(@s) + N'.' + QUOTENAME(@t)
                 + N' WHERE ' + @pred + N';';

        EXEC sp_executesql
             @sql,
             N'@fn nvarchar(200), @em nvarchar(320), @ph nvarchar(50), @n bigint OUTPUT',
             @fn = @fn, @em = @em, @ph = @ph, @n = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (CAST(@s + N'.' + @t AS nvarchar(300)), @cnt);

    FETCH NEXT FROM c INTO @oid, @s, @t;
END

CLOSE c;
DEALLOCATE c;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
DROP TABLE #targets;
```
