```sql
SET NOCOUNT ON;

CREATE TABLE #results
(
    object_name nvarchar(400) NOT NULL,
    rows_found  bigint        NOT NULL
);

-- LIKE patterns for the subject's original personal data ('_' escaped with '\')
DECLARE @p1 nvarchar(200) = N'%Subject\_0000042%',
        @p2 nvarchar(200) = N'%subject42@example.invalid%',
        @p3 nvarchar(200) = N'%+1-555-0000042%';

DECLARE @schema sysname, @table sysname, @obj nvarchar(400),
        @where nvarchar(max), @sql nvarchar(max), @cnt bigint;

DECLARE tbl CURSOR LOCAL FAST_FORWARD FOR
    SELECT s.name, t.name
    FROM sys.tables AS t
    JOIN sys.schemas AS s ON s.schema_id = t.schema_id
    WHERE t.is_ms_shipped = 0
    ORDER BY s.name, t.name;

OPEN tbl;
FETCH NEXT FROM tbl INTO @schema, @table;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @obj = QUOTENAME(@schema) + N'.' + QUOTENAME(@table);
    SET @where = NULL;
    SET @cnt = 0;

    SELECT @where = STRING_AGG(CAST(
        N'(CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max)) LIKE @p1 ESCAPE ''\'''
      + N' OR CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max)) LIKE @p2 ESCAPE ''\'''
      + N' OR CAST(' + QUOTENAME(c.name) + N' AS nvarchar(max)) LIKE @p3 ESCAPE ''\'')'
      AS nvarchar(max)), N' OR ')
    FROM sys.columns AS c
    JOIN sys.types AS ty ON ty.user_type_id = c.user_type_id
    WHERE c.object_id = OBJECT_ID(@obj)
      AND ty.name IN (N'char', N'varchar', N'nchar', N'nvarchar', N'text', N'ntext', N'xml');

    IF @where IS NOT NULL
    BEGIN
        SET @sql = N'SELECT @c = COUNT_BIG(*) FROM ' + @obj + N' WHERE ' + @where;

        EXEC sp_executesql
            @sql,
            N'@p1 nvarchar(200), @p2 nvarchar(200), @p3 nvarchar(200), @c bigint OUTPUT',
            @p1 = @p1, @p2 = @p2, @p3 = @p3, @c = @cnt OUTPUT;
    END

    INSERT INTO #results (object_name, rows_found)
    VALUES (@schema + N'.' + @table, @cnt);

    FETCH NEXT FROM tbl INTO @schema, @table;
END

CLOSE tbl;
DEALLOCATE tbl;

SELECT object_name, rows_found
FROM #results
ORDER BY object_name;

DROP TABLE #results;
```
